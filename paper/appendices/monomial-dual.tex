\section{Derivation of the monomial dual}
\label{app:dual-calculation}

\subsection{Generator action on monomials}
\label{app:generator-action}

Fix \(A\Subset\Lambda\). If \(i\notin A\), then
\(\chi_A(\eta^i)=\chi_A(\eta)\). If \(i\in A\), then
\begin{equation}
\label{eq:single-site-monomial-action}
\begin{aligned}
c_i(\eta)
\bigl(\chi_A(\eta^i)-\chi_A(\eta)\bigr)
={}&
c_i^0(\eta)\chi_{A\setminus\{i\}}(\eta)
\\
&-
\bigl(c_i^0(\eta)+c_i^1(\eta)\bigr)\chi_A(\eta).
\end{aligned}
\end{equation}
Indeed, when \(\eta(i)=0\), the left-hand side is
\(c_i^0(\eta)\chi_{A\setminus\{i\}}(\eta)\); when \(\eta(i)=1\), it is
\(-c_i^1(\eta)\chi_{A\setminus\{i\}}(\eta)\).

Insert the expansions \eqref{eq:rate-expansion} and the signed coefficients
\eqref{eq:signed-dual-coefficients}. Since
\[
\chi_S\chi_{A\setminus\{i\}}
=
\chi_{(A\setminus\{i\})\cup S},
\qquad
\chi_S\chi_A=\chi_{A\cup S},
\]
we obtain
\begin{equation}
\label{eq:generator-action-monomials}
\Lcal\chi_A
=
\sum_{i\in A}\sum_{S\subseteq N(i)}
a_i^\delta(S)\chi_{(A\setminus\{i\})\cup S}
+
\sum_{i\in A}\sum_{S\subseteq N(i)}
a_i^\beta(S)\chi_{A\cup S}.
\end{equation}
The terms with \(S=\varnothing\) in the first sum are deaths; the terms with
\(S\neq\varnothing\) are splits. The nonempty-target terms in the second sum
are births. Its empty-target term is diagonal because
\(A\cup\varnothing=A\).

\subsection{Dual rates, signs, and potential}
\label{app:dual-data}

Let \(Y=(A,\sigma)\) and let \(H\) be given by
\eqref{eq:monomial-duality-function}. By
\eqref{eq:death-split-data}--\eqref{eq:birth-data}, multiplying a rate by its
sign recovers the corresponding signed coefficient. Therefore
\begin{equation}
\label{eq:dual-generator-on-H}
\begin{aligned}
\Dcal H(Y,\eta)
={}&
\sigma
\sum_{i\in A}\sum_{S\subseteq N(i)}
a_i^\delta(S)
\chi_{(A\setminus\{i\})\cup S}(\eta)
\\
&+
\sigma
\sum_{i\in A}
\sum_{\substack{S\subseteq N(i)\\S\neq\varnothing}}
a_i^\beta(S)
\chi_{A\cup S}(\eta)
\\
&-
\left(
\sum_{i\in A}\sum_{S\subseteq N(i)}\delta_i(S)
+
\sum_{i\in A}
\sum_{\substack{S\subseteq N(i)\\S\neq\varnothing}}\beta_i(S)
\right)H(Y,\eta).
\end{aligned}
\end{equation}
By \eqref{eq:dual-potential}, the coefficient of \(H(Y,\eta)\) added by
\(V(A)H(Y,\eta)\), after cancelling the two total jump rates in
\eqref{eq:dual-generator-on-H}, is
\[
\sum_{i\in A}a_i^\beta(\varnothing).
\]
Consequently, \(\Dcal H+VH\) equals \(\sigma\) times the right-hand side of
\eqref{eq:generator-action-monomials}. This proves the generator identity
\eqref{eq:monomial-generator-duality}. The standard Feynman--Kac formula then
gives \eqref{eq:monomial-feynman-kac-duality}.
