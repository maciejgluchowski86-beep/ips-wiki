\subsection{Local finiteness and backward predecessors}
\label{app:local-finiteness}

Put
\begin{equation*}
\overline c
=
\sup_{i\in\Lambda,\,\eta\in\Omega}c_i(\eta),
\qquad
m_0=\sup_{i\in\Lambda}\abs{N(i)}.
\end{equation*}
M\"obius inversion in~\eqref{eq:rate-expansion} gives
\begin{equation*}
\abs{c_i^x(S)}
\leq
2^{\abs S}\overline c
\leq
2^{m_0}\overline c.
\end{equation*}
Hence
\begin{equation*}
\overline\alpha
=
\sup_{i\in\Lambda}
\bb{
 \sum_{S\subseteq N(i)}\delta_i(S)
 +
 \sum_{\substack{S\subseteq N(i)\\S\neq\vn}}\beta_i(S)
}
<\infty,
\end{equation*}
because~$\delta_i(S)=\abs{c_i^0(S)}$ and
$\beta_i(S)=\abs{c_i^0(S)+c_i^1(S)}$.

\begin{lemma}[Local finiteness]
\label{lem:dual-local-finiteness}
If the signed dual starts from~$A\Subset\Lambda$, then almost surely only
finitely many relevant interactions occur on every bounded time interval.
Consequently,~$\mathcal I_T$ and~$\mathcal P_T$ are finite for every~$T<\infty$,
interaction times are distinct, and every maximal backward predecessor chain
of patches with outgoing terminal boundary reaches time~$0$ after finitely many
steps.
\end{lemma}

\begin{proof}
When the active set is~$B$, the total rate of relevant clocks is at most
$\overline\alpha\abs B$. One interaction adds at most~$m_0$ sites. Hence the
number of active sites and the number of relevant rings are dominated by a
continuous-time branching process in which each particle rings at rate
$\overline\alpha$ and creates at most~$m_0$ new particles. This branching
process is nonexplosive. Independent Poisson processes have no simultaneous
points almost surely, and a countable union over pairs of clocks preserves this
property. The remaining conclusions follow from the definitions of the
successful skeleton and the patch family.
\end{proof}

In particular,
\begin{equation}
\label{eq:finite-patch-family-last-interaction}
\cb{\abs{\mathcal P}<\infty}
=
\bigcup_{n\in\N}L_n.
\end{equation}
Indeed, a finite patch family contains only finitely many successful
interactions and therefore has a last one. Conversely, on~$L_T$ there are only
finitely many interactions before~$T$ and none after it.

\subsection{Confined-interaction identity}
\label{app:confined-interactions}

For~$R\subseteq\Lambda$, let~$\eta^{R,0}$ agree with~$\eta$ on~$R$ and vanish on
$R^c$. Define
\begin{equation*}
c_{i,R}(\eta)
=
\pw{c_i\bb{\eta^{R,0}}}{i\in R}
{c_i\bb{\eta^{\cb{i},0}}}{i\notin R},
\end{equation*}
and let~$P_t^R$ be the corresponding semigroup. Inside~$R$, this is the
original spin system with zero boundary; sites outside~$R$ evolve independently.
Hence, if~$f$ depends only on spins in~$R$,
\begin{equation}
\label{eq:modified-zero-boundary-agreement}
P_t^Rf=P_t^{R,0}f.
\end{equation}

\begin{proposition}[Confined-interaction identity]
\label{prop:confined-interaction-identity}
Let~$A\subseteq R\Subset\Lambda$,~$0\leq T\leq t$, and let~$\mu$ be any
probability measure. Then
\begin{equation}
\label{eq:confined-interaction-identity}
\E_A\Cb{W_t^\mu\ind\bb{E_T^R}}
=
(\mu P_{t-T}P_T^{R,0})(\chi_A).
\end{equation}
\end{proposition}

\begin{proof}
For a fixed configuration~$\eta$, write
\begin{equation*}
Z_t^\eta
=
\sigma_t
\exp\bb{\int_0^tV(A_s)\,ds}
\chi_{A_t}(\eta).
\end{equation*}
The pathwise patch factorization and Theorem~\ref{thm:patch-factorization} give
$W_t^{\delta_\eta}=\CE{Z_t^\eta}{\mathcal G_t}$. Since~$E_T^R\in\mathcal G_t$,
\begin{equation}
\label{eq:confined-weight-to-dual}
\E_A\Cb{W_t^{\delta_\eta}\ind\bb{E_T^R}}
=
\E_A\Cb{Z_t^\eta\ind\bb{E_T^R}}.
\end{equation}

During the dual interval~$[0,T]$, retain every empty-target death interaction
and retain a nonempty-target split or birth only when its source and all its
targets lie in~$R$. Let~$\Dcal_R$ be the resulting signed-dual generator. If
the active set is~$B$, the total rate of discarded interactions is
\begin{equation*}
\kappa_R(B)
=
\sum_{i\in B}
\sum_{\substack{\vn\neq S\subseteq N(i)\\i\notin R\text{ or }S\nsubseteq R}}
\bb{\delta_i(S)+\beta_i(S)}.
\end{equation*}
The indicator of~$E_T^R$ kills the dual at the first discarded successful
interaction, so the Feynman--Kac operator on~$[0,T]$ is
$\Dcal_R+V-\kappa_R$.
{\color{red} Again problem with you describing steps instead of just showing them. Just show the duality between generator of the dual process without interactions reaching out of R and the spin system with 0 conditions outside R.}
This is the monomial-dual operator of the modified rates~$c_{i,R}$. For
$i\in R$, the multilinear expansion of~$c_{i,R}^x$ retains precisely the
coefficients~$c_i^x(S)$ with~$S\subseteq R$; for~$i\notin R$, only the
empty-neighbour coefficient remains. Thus the nonempty-target transitions are
those retained in~$\Dcal_R$, every empty-target death remains, and the discarded
rate subtracts~$\kappa_R$ from the potential.

Apply modified duality on~$[0,T]$ and original duality on~$[T,t]$. Dual
intervals act from right to left on observables, giving
\begin{equation*}
\E_A\Cb{Z_t^\eta\ind\bb{E_T^R}}
=
\bb{P_{t-T}P_T^R\chi_A}(\eta).
\end{equation*}
Since~$A\subseteq R$,~$\chi_A$ depends only on~$R$, and
\eqref{eq:modified-zero-boundary-agreement} replaces~$P_T^R$ by
$P_T^{R,0}$. Integrating over~$\mu$ proves
\eqref{eq:confined-interaction-identity}.
\end{proof}

&=

\label{eq:appendix-infinite-continuation-identity}
\end{align}
The same finite-horizon identity holds with every contribution replaced by its

\end{proposition}

\begin{proof}
For~$\eta\in\Omega$, use~$Z_t^\eta$ from the proof of
Proposition~\ref{prop:confined-interaction-identity}. The patch representation
gives~$W_t^{\delta_\eta}=\CE{Z_t^\eta}{\mathcal G_t}$, and
$L_{T,t}\in\mathcal G_t$. The tower property gives
\begin{equation*}
\CE{W_t^{\delta_\eta}\ind\bb{L_{T,t}}}{\mathcal G_T}
=
\CE{Z_t^\eta\ind\bb{L_{T,t}}}{\mathcal G_T}.
\end{equation*}

Conditional on~$\mathcal G_T$, the hidden data of the bulk patches have their
consistent patch laws. Future Poisson increments based at the distinct sites
of~$\mathcal E_T$ are independent of these data and of one another. While no
successful interaction occurs, no site outside~$\mb{Cone}_T$ becomes
dual-active. Hence~$L_{T,t}$ is the intersection of the local events that the
continuations of the patches in~$\mathcal E_T$ create no successful interaction
before~$t$. The conditional product disintegration is
\begin{align*}
\CE{Z_t^\eta\ind\bb{L_{T,t}}}{\mathcal G_T}
&=
\prod_{P\in\mathcal B_T}
\E_P^{\mathrm{con}}\Cb{F(P)}
\prod_{P\in\mathcal E_T}
C\bb{\eta\bb{i(P)},P^{\uparrow t}}
\\
&=
\prod_{P\in\mathcal B_T}C(P)
\prod_{P\in\mathcal E_T}
C\bb{\eta\bb{i(P)},P^{\uparrow t}}.
\end{align*}
The continuation factor includes the local no-success event by
\eqref{eq:end-patch-continuation}. Integrating over~$\eta$ proves the first
identity. The proof for~$\Lcal^\varepsilon$ is identical.

On~$L_T$, the patch family consists of~$\mathcal B_T$ and one infinite
continuation of each patch in~$\mathcal E_T$. The same disintegration over the
independent future increments gives one factor~$C(P^{\uparrow\infty})$ per end
patch. These factors are the limits in~\eqref{eq:end-patch-continuation}, which
proves the second identity.
\end{proof}

The next lemma contains the only algebra needed for the no-late estimate, comparing difference of products via a telescoping identity.

\begin{lemma}[End-factor relaxation]
\label{lem:end-factor-relaxation}
Fix a realization of~$\mathcal G_T$ and let~$0\leq T<t$. For every
$\mu\in\mathcal M_*$,
\begin{align*}
&\abs{
 \mu\bb{
  \prod_{P\in\mathcal E_T}
  C\bb{\eta\bb{i(P)},P^{\uparrow t}}
 }
 -
 \prod_{P\in\mathcal E_T}C\bb{P^{\uparrow\infty}}
}
\\
&\qquad\leq
e^{-\varepsilon\bb{t-T}}
\bb{
 \mu\bb{
  \prod_{P\in\mathcal E_T}
  C^\varepsilon\bb{\eta\bb{i(P)},P^{\uparrow t}}
 }
 +
 \abs{\mathcal E_T}
 \prod_{P\in\mathcal E_T}C\bb{P^{\uparrow\infty}}
}.
\end{align*}
\end{lemma}

\begin{proof}
Put~$h=t-T$. For~$P\in\mathcal E_T$ based at~$i$, write
\begin{align*}
a_t(P)&=C\bb{p_i^\star,P^{\uparrow t}},
&k_t(P)&=\partial_zC\bb{z,P^{\uparrow t}},
\\
a_t^\varepsilon(P)&=C^\varepsilon\bb{p_i^\star,P^{\uparrow t}},
&k_t^\varepsilon(P)&=\partial_zC^\varepsilon\bb{z,P^{\uparrow t}},
\\
b(P)&=C\bb{P^{\uparrow\infty}}.
\end{align*}
Let~$r_i=c_i^0(\vn)+c_i^1(\vn)$ and
$q_i=c_i^0(\vn)/r_i$. Uniform deaths give~$r_i\geq\varepsilon$, while
Corollary~\ref{cor:critical-profile-empty-bound} gives~$p_i^\star\leq q_i$.
The continuation formula and pure-death comparison imply
\begin{equation}
\label{eq:appendix-continued-factor-comparisons}
0\leq a_t(P)\leq a_t^\varepsilon(P),
\qquad
0\leq k_t(P)\leq e^{-\varepsilon h}k_t^\varepsilon(P),
\qquad
0\leq b(P)-a_t(P)\leq e^{-\varepsilon h}b(P).
\end{equation}
Indeed,
\begin{equation*}
k_t(P)
=
e^{-r_i h}\kappa(P)
\leq
e^{-\varepsilon h}e^{-\bb{r_i-\varepsilon}h}\kappa^\varepsilon(P)
=
e^{-\varepsilon h}k_t^\varepsilon(P),
\end{equation*}
and
\begin{equation*}
b(P)-a_t(P)
=
\kappa(P)\bb{q_i-p_i^\star}e^{-r_i h}
\leq
e^{-\varepsilon h}b(P),
\end{equation*}
because~$b(P)=C(p_i^\star,P)+\kappa(P)(q_i-p_i^\star)$.

For~$\mathcal Q\subseteq\mathcal E_T$, put
$I(\mathcal Q)=\cb{i(P):P\in\mathcal Q}$. The centered expansion gives
\begin{equation*}
\mu\bb{
 \prod_{P\in\mathcal E_T}
 C\bb{\eta\bb{i(P)},P^{\uparrow t}}
}
=
\sum_{\mathcal Q\subseteq\mathcal E_T}
\mu\bb{\chi_{I(\mathcal Q)}^*}
\prod_{P\in\mathcal Q}k_t(P)
\prod_{P\in\mathcal E_T\setminus\mathcal Q}a_t(P).
\end{equation*}
All terms are nonnegative. For~$\mathcal Q\neq\vn$, the first two inequalities
in~\eqref{eq:appendix-continued-factor-comparisons} bound the corresponding
term by~$e^{-\varepsilon h}$ times its~$\varepsilon$ counterpart. Therefore
\begin{align*}
0
&\leq
\mu\bb{
 \prod_{P\in\mathcal E_T}
 C\bb{\eta\bb{i(P)},P^{\uparrow t}}
}
-
\prod_{P\in\mathcal E_T}a_t(P)
\\
&\leq
e^{-\varepsilon h}
\mu\bb{
 \prod_{P\in\mathcal E_T}
 C^\varepsilon\bb{\eta\bb{i(P)},P^{\uparrow t}}
}.
\end{align*}

Enumerate~$\mathcal E_T=\cb{P_1,\ldots,P_n}$ and write
$a_j=a_t(P_j)$ and~$b_j=b(P_j)$. The telescoping identity
\begin{equation*}
\prod_{j=1}^n b_j-
\prod_{j=1}^n a_j
=
\sum_{j=1}^n
\bb{b_j-a_j}
\prod_{k<j}a_k
\prod_{k>j}b_k
\end{equation*}
and the last inequality in
\eqref{eq:appendix-continued-factor-comparisons} give
\begin{equation*}
0
\leq
\prod_{P\in\mathcal E_T}b(P)
-
\prod_{P\in\mathcal E_T}a_t(P)
\leq
\abs{\mathcal E_T}e^{-\varepsilon h}
\prod_{P\in\mathcal E_T}b(P).
\end{equation*}
Both target quantities lie above the common product
$\prod_Pa_t(P)$, so the sum of the two preceding errors bounds their absolute
difference.
\end{proof}

\subsection{Finite-volume and finite-propagation approximations}
\label{app:finite-propagation}

Put
\begin{equation*}
m=\sup_{i\in\Lambda}\abs{N(i)\cup\cb{i}}<\infty.
\end{equation*}

\begin{lemma}[Finite propagation]
\label{lem:finite-propagation}
{\color{red} This is standard, give appropriate citation and say that the proof is given to make paper self contained.}
Let~$f$ be supported on~$A\Subset\Lambda$. For every~$a>0$, there are
$v<\infty$ and~$C_A<\infty$ such that
\begin{equation}
\label{eq:finite-propagation}
\norm{\bb{P_t-P_t^{R,0}}f}_\infty
\leq
C_Ae^{-at}\norm f_\infty
\end{equation}
for every~$t\geq0$ and every finite~$R$ containing
$B\bb{A,\lceil vt\rceil}$.
\end{lemma}

\begin{proof}
Construct the process from rate-$\overline c$ candidate clocks and independent
acceptance marks. Explore the clocks backwards from~$A$ at time~$t$. Whenever
a candidate clock at~$i$ is encountered, add~$N(i)\cup\cb{i}$ to the sites whose
earlier histories must be known. If this backward influence cluster stays
inside~$R$, the infinite-volume and zero-boundary processes, coupled with the
same clocks and marks in~$R$, agree on~$A$ at time~$t$.

To leave~$B(A,n)$, the cluster must contain a chronological dependency path of
at least~$n$ candidate clocks. There are at most~$\abs A m^k$ spatial paths of
length~$k$, and the expected number of chronologically ordered~$k$-tuples along
a fixed path is~$(\overline c t)^k/k!$. Therefore, for every~$\theta>0$,
\begin{equation*}
\pP\bb{\text{the influence cluster leaves }B(A,n)}
\leq
\abs A\sum_{k\geq n}\frac{\bb{m\overline c t}^k}{k!}
\leq
\abs A\exp\bb{m\overline c t e^\theta-\theta n}.
\end{equation*}
Take~$n=\lceil vt\rceil$ and choose~$v$ so that
$\theta v-m\overline c e^\theta>a$. After increasing the constant to cover
bounded~$t$, the exit probability is at most~$C_Ae^{-at}$. The graphical
coupling gives
\begin{equation*}
\abs{P_tf(\eta)-P_t^{R,0}f(\eta)}
\leq
2\norm f_\infty
\pP\bb{\text{the influence cluster leaves }R},
\end{equation*}
uniformly in~$\eta$, and proves~\eqref{eq:finite-propagation}.
\end{proof}

For fixed~$t$ and local~$f$, Lemma~\ref{lem:finite-propagation} implies uniform
convergence of zero-boundary finite-volume semigroups to~$P_tf$ along any
exhaustion whose boundary recedes from the dependence set of~$f$. In
particular, the nonnegative centered-monomial coefficients of the finite-volume
semigroups used in Theorem~\ref{thm:centered-moment-order} pass to the
infinite-volume limit.
