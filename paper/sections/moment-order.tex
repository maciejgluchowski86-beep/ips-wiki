\subsection{Centered-moment order}
\label{sec:moment-order}
\label{subsec:moment-monotonicity}

We now prove part~\textup{(ii)} of Theorem~\ref{thm:patch-positivity-order}.
We write the generator in the centered-monomial basis and show that its
off-diagonal coefficients are nonnegative. The corresponding equations for the
centered moments then form a row-finite Metzler system, so the standard
positivity theorem for Metzler generators~\cite{ArendtEtAl1986} gives
preservation of nonnegative centered moments.

If~$i\notin A$, the update at~$i$ leaves~$\chi_A^*$ unchanged. If~$i\in A$,
then
\begin{equation*}
\lambda_i\bb{E_{i,\rho_i}-I}\chi_A^*
=
\lambda_i\bb{\rho_i-p_i^\star}\chi_{A\setminus\cb{i}}^*
-
\lambda_i\chi_A^*.
\end{equation*}
Expanding~$\lambda_i\rho_i$ and~$\lambda_i$ in monomials and summing over
$i\in A$ gives
\begin{equation*}
\Lcal\chi_A^*
=\sum_{i\in A}\Cb{
-\lambda_i(\mb 0)\chi_A^*
+
\sum_{S\subseteq N(i)}
\bb{
\widehat{\lambda_i\rho_i}(S)
-p_i^\star\widehat{\lambda_i}(S)
}
\chi_S\chi_{A\setminus\cb{i}}^*
+
\sum_{\substack{S\subseteq N(i)\\S\neq\vn}}
\bb{-\widehat{\lambda_i}(S)}
\chi_S\chi_A^*
}.
\end{equation*}
Part~\textup{(i)} gives
\begin{equation*}
-\widehat{\lambda_i}(S)\geq0
\quad\text{for }S\neq\vn,
\qquad
\widehat{\lambda_i\rho_i}(S)
-p_i^\star\widehat{\lambda_i}(S)
\geq0
\quad\text{for every }S\subseteq N(i).
\end{equation*}
For the second inequality, the case~$S=\vn$ is
$\lambda_i(\mb 0)(\rho_i(\mb 0)-p_i^\star)\geq0$. If~$S\neq\vn$ and
$\widehat{\lambda_i}(S)<0$, it follows from the definition of~$p_i^\star$; if
$\widehat{\lambda_i}(S)=0$, it follows from the second coefficient inequality
in part~\textup{(i)}.

It remains to expand the products~$\chi_S\chi_B^*$ in the centered basis. For
finite~$S,B\subseteq\Lambda$,
\begin{equation*}
\chi_S\chi_B^*
=
\bb{\prod_{j\in S\cap B}\bb{1-p_j^\star}}
\sum_{R\subseteq S}
\bb{\prod_{j\in S\setminus R}p_j^\star}
\chi_{R\cup(B\setminus S)}^*.
\end{equation*}
Indeed, first use
$\eta(j)(\eta(j)-p_j^\star)=(1-p_j^\star)\eta(j)$ on~$S\cap B$, and then
expand~$\eta(j)=p_j^\star+(\eta(j)-p_j^\star)$ for~$j\in S$. Since
$0\leq p_j^\star\leq1$, every coefficient is nonnegative. Hence, after
collecting equal centered monomials,
\begin{equation*}
\Lcal\chi_A^*
=
\sum_{B\Subset\Lambda}M(A,B)\chi_B^*,
\qquad
M(A,B)\geq0\quad\text{for }B\neq A,
\end{equation*}
and for each~$A$ only finitely many coefficients~$M(A,B)$ are nonzero.

Let~$\sigma$ be a finite signed measure and put
$m_A(t)=(\sigma P_t)(\chi_A^*)$. The backward equation gives
\begin{equation*}
\frac{d}{dt}m_A(t)
=
\sum_{B\Subset\Lambda}M(A,B)m_B(t).
\end{equation*}
This is a row-finite Metzler system. By the positivity result cited above,
$m_A(0)\geq0$ for every finite~$A$ implies~$m_A(t)\geq0$ for every finite~$A$
and~$t\geq0$. Taking~$\sigma=\nu-\mu$ therefore proves
\begin{equation*}
\mu\preceq_*\nu
\quad\Longrightarrow\quad
\mu P_t\preceq_*\nu P_t
\qquad\text{for every }t\geq0.
\end{equation*}

The same argument with~$\sigma=\mu$ shows that~$\mathcal M_*$ is preserved by
the semigroup. Ordinary monomial comparison follows from expansion
\begin{equation*}
\chi_A
=
\sum_{B\subseteq A}
\bb{\prod_{i\in A\setminus B}p_i^\star}\chi_B^*,
\end{equation*}
whose coefficients are nonnegative. Finally, every nonempty centered moment of
$\mu_{\mb p^\star}$ is zero, so
$\mu_{\mb p^\star}\preceq_*\mu$ for every~$\mu\in\mathcal M_*$. Applying the
monomial comparison gives
\begin{equation*}
(\mu_{\mb p^\star}P_t)(\chi_A)
\leq
(\mu P_t)(\chi_A).
\end{equation*}
This proves part~\textup{(ii)} and completes the proof of
Theorem~\ref{thm:patch-positivity-order}.

\subsection{Pure-death comparison}
\label{subsec:pure-death-comparison}

Let~$\Lcal$ and~$\Lcal'$ satisfy
\begin{equation*}
\Lcal f(\eta)
=
\Lcal'f(\eta)
+
\sum_{i\in\Lambda}d_i
\bb{f\bb{\eta^{i,0}}-f(\eta)},
\qquad d_i\geq0,
\end{equation*}
as in Corollary~\ref{cor:pure-death-comparison}. Thus~$\Lcal$ has the additional
pure deaths and~$\Lcal'$ is obtained by removing them. We prove that, for every
$\mu\in\mathcal M_*$,
\begin{equation*}
(\mu P_t)(\chi_A)
\leq
(\mu P_t')(\chi_A).
\end{equation*}

Recall the duality function
$H((A,\sigma),\eta)=\sigma\chi_A(\eta)$. The pure-death term acts diagonally:
\begin{equation*}
\sum_{i\in\Lambda}d_i\bb{E_{i,0}-I}H((A,\sigma),\eta)
=
-\bb{\sum_{i\in A}d_i}H((A,\sigma),\eta).
\end{equation*}
Hence, if
\begin{equation*}
\Lcal' H
=
\Dcal' H+V'(A)H,
\end{equation*}
is the signed-duality relation for~$\Lcal'$, then
\begin{equation*}
\Lcal H
=
\Dcal' H
+
\bb{V'(A)-\sum_{i\in A}d_i}H.
\end{equation*}
Thus the two systems have the same signed dual generator,
$\Dcal=\Dcal'$, while their one-site Feynman--Kac potentials satisfy
$V_i=V_i'-d_i$. In particular, they have the same successful-interaction
skeleton, the same patch families~$\mathcal B_t$ and~$\mathcal E_t$, and the
same consistent patch laws. Write~$C,C'$ for their patch contributions. The
patch representation gives, with the same expectation~$\E_A$ in both lines,
\begin{align*}
(\mu P_t)(\chi_A)
&=
\E_A\Cb{
\prod_{P\in\mathcal B_t}C(P)\,
\mu\bb{
\prod_{P\in\mathcal E_t}C\bb{\eta\bb{i(P)},P}
}
},
\\
(\mu P_t')(\chi_A)
&=
\E_A\Cb{
\prod_{P\in\mathcal B_t}C'(P)\,
\mu\bb{
\prod_{P\in\mathcal E_t}C'\bb{\eta\bb{i(P)},P}
}
}.
\end{align*}

The explicit formulas in
Appendix~\ref{app:contribution-formulas}, together with
$V_i=V_i'-d_i$ and the patch-positivity inequality
$-\widehat{\lambda_i}(S)\geq0$ for~$S\neq\vn$, give
\begin{equation*}
0\leq C(P)\leq C'(P),
\qquad
C(z,P)\leq C'(z,P)
\quad\text{for }z\in[0,1].
\end{equation*}
Thus~$\Lcal'$ is patch-positive. Moreover,
\eqref{eq:pure-death-decomposition} gives
$\lambda_i=\lambda_i'+d_i$ and
$\lambda_i\rho_i=\lambda_i'\rho_i'$. Hence the nonempty M\"obius coefficients
in~\eqref{eq:principal-threshold-profile} are the same for the two systems, so
they have the same patch threshold profile~$\mb p^\star$.

For an end patch~$P$ based at~$i$, set
$\kappa(P)=\partial_zC(z,P)$ and
$\kappa'(P)=\partial_zC'(z,P)$. The same formulas give
\begin{equation*}
0\leq\kappa(P)\leq\kappa'(P),
\qquad
0\leq C\bb{p_i^\star,P}
\leq C'\bb{p_i^\star,P}.
\end{equation*}
Since~$C(z,P)$ is affine in~$z$,
\begin{equation*}
C\bb{\eta(i),P}
=
C\bb{p_i^\star,P}
+
\kappa(P)\bb{\eta(i)-p_i^\star}.
\end{equation*}
Distinct end patches are based at distinct sites, so
\begin{equation*}
\mu\bb{
\prod_{P\in\mathcal E_t}C\bb{\eta\bb{i(P)},P}
}
=
\sum_{\mathcal Q\subseteq\mathcal E_t}
\mu\bb{\chi_{\cb{i(P):P\in\mathcal Q}}^*}
\prod_{P\in\mathcal Q}\kappa(P)
\prod_{P\in\mathcal E_t\setminus\mathcal Q}
C\bb{p_{i(P)}^\star,P}.
\end{equation*}
The same expansion holds with~$C$ and~$\kappa$ replaced by~$C'$ and~$\kappa'$.
If~$\mu\in\mathcal M_*$, every centered moment in the sum is nonnegative, and
the coefficient inequalities above therefore imply
\begin{equation*}
\mu\bb{
\prod_{P\in\mathcal E_t}C\bb{\eta\bb{i(P)},P}
}
\leq
\mu\bb{
\prod_{P\in\mathcal E_t}C'\bb{\eta\bb{i(P)},P}
}.
\end{equation*}
Together with~$0\leq C(P)\leq C'(P)$ for the bulk patches, the two patch
representations give
\begin{equation*}
(\mu P_t)(\chi_A)
\leq
(\mu P_t')(\chi_A).
\end{equation*}
This proves Corollary~\ref{cor:pure-death-comparison}.

\begin{corollary}[Invariant measure comparison]
\label{cor:pure-death-invariant-comparison}
Under the hypotheses of Corollary~\ref{cor:pure-death-comparison}, suppose that
the two systems have unique invariant measures~$\pi$ and~$\pi'$. Then
\begin{equation*}
\pi(\chi_A)\leq\pi'(\chi_A)
\qquad
\text{for every }A\Subset\Lambda.
\end{equation*}
\end{corollary}

\begin{proof}
Apply Corollary~\ref{cor:pure-death-comparison} with
$\mu=\mu_{\mb 1}\in\mathcal M_*$ and average over~$t\in[0,T]$. Every
subsequential weak limit of either family of Ces\`aro averages is invariant.
Uniqueness identifies the limits as~$\pi$ and~$\pi'$, and letting~$T\to\infty$
gives the claim.
\end{proof}
