\section{Patch laws and representation}
\label{sec:representation}

\subsection{Reference and consistent patch laws}
\label{subsec:patch-laws}

Fix a labeled patch~$P$, either a full patch in~$\mathcal P$ or a patch
in~$\mathcal P_T$ for some finite~$T$. For brevity, write
\begin{equation*}
i=i(P),
\qquad
s=s(P),
\qquad
e=e(P),
\qquad
S=S(P).
\end{equation*}
The interior interaction data of~$P$ are the marked interactions outgoing
from~$i$ strictly between its endpoints:
\begin{equation*}
\Sigma_P^\circ
=
\cb{(i,u,\gamma,R)\in I:s<u<e}.
\end{equation*}
Under the reference law, the type-$(\delta,R)$ and type-$(\beta,R)$ points in
this interval are independent Poisson processes with the rates from
Section~\ref{subsec:signed-set-process}.

If~$\mathsf X(P)=\mathsf I$, the initial interaction of~$P$ is incoming from
another source, and we set~$\Sigma_P=\Sigma_P^\circ$. Let~$\pP_P$ be the
corresponding product Poisson law. If~$\mathsf X(P)=\mathsf O$, the initial
interaction is outgoing from~$i$, with target~$S$, but its kind is not revealed
by the skeleton. Let~$\alpha(P)\in\cb{\delta,\beta}$ be this hidden kind,
independent of~$\Sigma_P^\circ$, with
\begin{equation}
\label{eq:patch-initial-kind-law}
\pP_P\bb{\alpha(P)=\delta}
=
\frac{\delta_i(S)}{\delta_i(S)+\beta_i(S)},
\qquad
\pP_P\bb{\alpha(P)=\beta}
=
\frac{\beta_i(S)}{\delta_i(S)+\beta_i(S)}.
\end{equation}
In this case,
\begin{equation*}
\Sigma_P
=
\Sigma_P^\circ\cup\cb{(i,s,\alpha(P),S)},
\end{equation*}
and~$\pP_P$ is the product of the law in
\eqref{eq:patch-initial-kind-law} and the interior Poisson law.
Write~$\E_P$ for expectation under~$\pP_P$.

On a realized global history, the local active indicator is
$X_u^P=\ind\cb{i(P)\in A_u}$ for~$s(P)\leq u<e(P)$. Under the reference patch
law, we reconstruct the candidate process~$X^P$ from~$\Sigma_P$ by the same
local update rules. Its initial value is
\begin{equation*}
X_s^P
=
\pw{1}{\mathsf X(P)=\mathsf I}
{\ind\cb{\alpha(P)=\beta}}{\mathsf X(P)=\mathsf O}.
\end{equation*}
Reading~$\Sigma_P^\circ$ chronologically, a~$\delta$-mark sends an active
source to~$0$, a~$\beta$-mark leaves an active source at~$1$, and every mark at
an inactive source is ignored. Between marks,~$X^P$ is constant. For a finite
patch with terminal label~$\mathsf E$, we also set~$X_e^P=X_{e-}^P$.

The candidate local history is \emph{consistent} with the prescribed skeleton
when it produces no additional successful interaction in the patch and, if the
patch ends with an outgoing interaction, makes that interaction successful.
Thus
\begin{equation*}
\operatorname{Con}(P)
=
\cb{
X_{u-}^P=0
\text{ for every }(i,u,\gamma,R)\in\Sigma_P^\circ
\text{ with }R\neq\varnothing
}
\cap
\pw{\cb{X_{e-}^P=1}}{\mathsf Y(P)=\mathsf O}
{\Omega_P}{\mathsf Y(P)\in\cb{\mathsf I,\mathsf E}},
\end{equation*}
where~$\Omega_P$ is the reference sample space. For every patch in a realized
patch family,~$\pP_P\bb{\operatorname{Con}(P)}>0$ almost surely. The consistent
patch law is
\begin{equation*}
\pP_P^{\mathrm{con}}(\cdot)
=
\pP_P\bb{\,\cdot\mid\operatorname{Con}(P)},
\end{equation*}
and~$\E_P^{\mathrm{con}}$ denotes its expectation.

\subsection{Conditional factorization}
\label{subsec:factorization}

The successful skeleton fixes the patch boundaries but leaves the local data
between them hidden. Conditioning on the complete skeleton imposes one local
consistency event on each disjoint source-time strip.

\begin{theorem}[Patch factorization]
\label{thm:patch-factorization}
Fix~$T<\infty$. Conditional on~$\mathcal G_T$, the interaction
data~$(\Sigma_P)_{P\in\mathcal P_T}$ are independent, and the conditional law
of~$\Sigma_P$ is~$\pP_P^{\mathrm{con}}$. Equivalently, for bounded measurable
functions~$f_P$,
\begin{equation}
\label{eq:patch-factorization}
\CE{
\prod_{P\in\mathcal P_T}f_P\bb{\Sigma_P}
}{\mathcal G_T}
=
\prod_{P\in\mathcal P_T}
\E_P^{\mathrm{con}}\Cb{f_P\bb{\Sigma_P}}.
\end{equation}
\end{theorem}

\begin{proof}
Write~$G_T=(Y_0,\mathcal I_T)$, so~$\mathcal G_T=\sigma(G_T)$, and condition on
a realization~$G_T=g$. The realization fixes the boundary interactions and a
finite labeled patch family~$\mathcal R=\mathcal P_T(g)$. Before requiring that
these boundary interactions form the complete successful skeleton, the raw
data in distinct patches have the product reference law
\begin{equation*}
\pP_g^0
=
\bigotimes_{P\in\mathcal R}\pP_P.
\end{equation*}
Indeed, the patch interiors are disjoint source-time regions, and the hidden
kind at each outgoing initial boundary has the rate-proportional law
\eqref{eq:patch-initial-kind-law}.

Relative to these fixed boundary interactions, the successful skeleton through
time~$T$ is~$g$ if and only if
\begin{equation*}
\bigcap_{P\in\mathcal R}\operatorname{Con}(P)
\end{equation*}
occurs. The forward implication follows from the definition of consistency. For
the reverse implication, read the boundary interactions chronologically. At
time zero, the incoming patches reproduce the prescribed active set. Between
consecutive boundary times, the local indicators agree with the global active
state, and consistency excludes every unrecorded successful interaction. At an
outgoing boundary, terminal consistency makes the source active, so the
boundary interaction is successful; its hidden kind gives the prescribed
post-boundary source state, while its targets begin the prescribed incoming
patches. Induction reconstructs the active process and the skeleton through
time~$T$.

Consequently, for bounded measurable~$f_P$,
\begin{align*}
\CE{
\prod_{P\in\mathcal R}f_P\bb{\Sigma_P}
}{G_T=g}
&=
\frac{
\displaystyle
\prod_{P\in\mathcal R}
\E_P\Cb{
f_P\bb{\Sigma_P}\ind_{\operatorname{Con}(P)}
}
}{
\displaystyle
\prod_{P\in\mathcal R}
\pP_P\bb{\operatorname{Con}(P)}
}
\\
&=
\prod_{P\in\mathcal R}
\E_P^{\mathrm{con}}\Cb{f_P\bb{\Sigma_P}}.
\end{align*}
The denominator is positive for almost every realized skeleton. This proves
\eqref{eq:patch-factorization}.
\end{proof}

\subsection{Patch contributions}
\label{subsec:patch-contributions}

The Feynman--Kac random variable in
\eqref{eq:monomial-feynman-kac-duality} factors into one local factor for each
patch. The sign assigned to the initial boundary of a patch is
\begin{equation*}
\sigma(P)
=
\pw{+}{\mathsf X(P)=\mathsf I}
{\sigma_{i(P)}^{\alpha(P)}\bb{S(P)}}{\mathsf X(P)=\mathsf O}.
\end{equation*}
For a full patch~$P\in\mathcal P$, define
\begin{equation*}
F(P)
=
\sigma(P)
\exp\bb{
V_{i(P)}\int_{s(P)}^{e(P)}X_u^P\,du
},
\qquad
C(P)=\E_P^{\mathrm{con}}\Cb{F(P)}.
\end{equation*}
When~$e(P)=\infty$, the exponential is interpreted by its limit over finite
upper endpoints; the limit always exists. For an end patch~$P\in\mathcal E_t$
and~$z\in[0,1]$, define
\begin{equation*}
F(z,P)
=
\sigma(P)
\exp\bb{
V_{i(P)}\int_{s(P)}^tX_u^P\,du
}
z^{X_t^P},
\qquad
C(z,P)=\E_P^{\mathrm{con}}\Cb{F(z,P)}.
\end{equation*}
The function~$z\mapsto C(z,P)$ is affine. The full and end contributions are
elementary one-site quantities; Appendix~\ref{app:contribution-formulas}
derives explicit formulas for every boundary type.

\subsection{The patch representation theorem}
\label{subsec:representation-theorem}

\begin{theorem}[Patch representation]
\label{thm:patch-representation}
For every~$A\Subset\Lambda$,~$t\geq0$, and~$\eta\in\Omega$,
\begin{equation}
\label{eq:patch-representation}
P_t\chi_A(\eta)
=
\E_A\Cb{
\prod_{P\in\mathcal B_t}C(P)
\prod_{P\in\mathcal E_t}C\bb{\eta\bb{i(P)},P}
}.
\end{equation}
\end{theorem}

\begin{proof}
On the global graphical space, let~$X^P$,~$\alpha(P)$, and~$\sigma(P)$ denote
the local data induced on each patch. Every nonempty-target successful
interaction contributes its sign to the unique patch which begins at its
source. Pure deaths have positive sign because
\begin{equation*}
a_i^\delta(\varnothing)
=
c_i^0(\varnothing)
=
c_i\bb{\mb{1}^{\,i}}
\geq0.
\end{equation*}
Hence
\begin{equation*}
\sigma_t
=
\prod_{P\in\mathcal B_t}\sigma(P)
\prod_{P\in\mathcal E_t}\sigma(P).
\end{equation*}
Since~$V(A_u)=\sum_{i\in A_u}V_i$, the patch intervals also split the
Feynman--Kac integral:
\begin{equation*}
\int_0^tV(A_u)\,du
=
\sum_{P\in\mathcal B_t}
V_{i(P)}\int_{s(P)}^{e(P)}X_u^P\,du
+
\sum_{P\in\mathcal E_t}
V_{i(P)}\int_{s(P)}^tX_u^P\,du.
\end{equation*}
The end patches record the active set at time~$t$, so
\begin{equation*}
\chi_{A_t}(\eta)
=
\prod_{P\in\mathcal E_t}
\eta\bb{i(P)}^{X_t^P}.
\end{equation*}
Combining these identities gives the pathwise factorization
\begin{equation}
\label{eq:pathwise-patch-factorization}
\sigma_t
\exp\bb{\int_0^tV(A_u)\,du}
\chi_{A_t}(\eta)
=
\prod_{P\in\mathcal B_t}F(P)
\prod_{P\in\mathcal E_t}F\bb{\eta\bb{i(P)},P}.
\end{equation}
Conditioning~\eqref{eq:pathwise-patch-factorization} on~$\mathcal G_t$ and
applying Theorem~\ref{thm:patch-factorization} replaces each local factor by
its consistent patch expectation. Taking expectation and using
Theorem~\ref{thm:monomial-duality} proves~\eqref{eq:patch-representation}.
\end{proof}

\subsection{General initial laws}
\label{subsec:general-initial-laws}

Let~$\mu$ be any probability measure on~$\Omega$. Integrating
\eqref{eq:patch-representation} gives
\begin{equation}
\label{eq:patch-representation-general-law}
\mu(P_t\chi_A)
=
\E_A\Cb{
\prod_{P\in\mathcal B_t}C(P)\,
\mu\bb{
\prod_{P\in\mathcal E_t}C\bb{\eta\bb{i(P)},P}
}
}.
\end{equation}
In particular, for the Bernoulli product measure~$\mu_{\mb p}$ with one-density
profile~$\mb p=(p_i)_{i\in\Lambda}$, the end-patch sites are distinct and the
end contributions are affine, so
\begin{equation}
\label{eq:patch-representation-product-law}
\mu_{\mb p}(P_t\chi_A)
=
\E_A\Cb{
\prod_{P\in\mathcal B_t}C(P)
\prod_{P\in\mathcal E_t}C\bb{p_{i(P)},P}
}.
\end{equation}
The skeleton retains the interactions which transmit dependence between sites.
Everything local between consecutive incoming or outgoing successful
interactions is integrated inside~$C(P)$ or~$C(z,P)$ before its sign is
estimated. Thus~\eqref{eq:patch-representation} is exact, but groups the signed
dual history in a way that preserves the cancellations hidden by termwise
absolute values.
