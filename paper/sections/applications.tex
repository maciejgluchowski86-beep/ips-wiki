We now apply our main results to several familiar spin systems. FA-1f is a
basic KCSM for which patch positivity holds, while higher-order FA models show
that patch positivity does not hold for all KCSMs. BABP gives an example
outside the KCSM class, and the contact process gives an example which is also
monotone in the usual configuration order. We then use FA-1f and BABP to
illustrate the centered-moment comparison and the convergence results proved
above.

\subsection{FA-1f}
\label{subsec:fa-1f-applications}

Fix~$p\in(0,1)$ and assume that~$N(i)\neq\vn$ for every~$i\in\Lambda$, so there are no isolated sites.
In FA-1f~\cite{FredricksonAndersen1984,CancriniEtAl2008,HartarskyToninelli2025},
state~$0$ is vacant and state~$1$ is occupied. A site is refreshed from the
Bernoulli law with parameter~$p$ whenever at least one neighbour is vacant, so
\begin{equation*}
\lambda_i(\eta)=1-\chi_{N(i)}(\eta),
\qquad
\rho_i(\eta)=p.
\end{equation*}
The only nonzero nonconstant M\"obius coefficient of~$\lambda_i$ is
\begin{equation*}
\widehat{\lambda_i}\bb{N(i)}=-1,
\end{equation*}
while~$\lambda_i\bb{\rho_i-\rho_i(\mb 0)}=0$. Hence
Theorem~\ref{thm:patch-positivity-order} shows that FA-1f is patch-positive.
Moreover,
\begin{equation*}
\frac{\widehat{\lambda_i\rho_i}\bb{N(i)}}
{\widehat{\lambda_i}\bb{N(i)}}=p,
\end{equation*}
so its patch threshold profile is~$\mb p^\star=\mb p$.

\subsection{FA-$j$f for~$j\geq2$}

The analogous conclusion fails for the higher-order FA models. In FA-$j$f,
a site is refreshed from the Bernoulli law with parameter~$p$ only when at
least~$j$ neighbours are vacant. Thus~$\rho_i=p$, while, writing
$m=\abs{N(i)}$,
\begin{equation*}
\lambda_i(\eta)
=
\ind\bb{\sum_{k\in N(i)}\bb{1-\eta(k)}\geq j}.
\end{equation*}
Suppose that~$2\leq j\leq m$, and let~$S\subseteq N(i)$ satisfy
$\abs S=m-j+2$. By M\"obius inversion,
\begin{equation*}
\widehat{\lambda_i}(S)
=
\sum_{r=0}^{\abs S-2}
(-1)^{\abs S-r}\binom{\abs S}{r}
=
\abs S-1
>
0.
\end{equation*}
This violates the first condition in
\eqref{eq:principal-coefficient-criterion}. Hence FA-$j$f is not
patch-positive for~$j\geq2$, even though its kinetic constraint is monotone
with respect to vacancies.

\begin{remark}[Contact process]
\label{subsec:contact-benchmark}

The contact process~\cite{Harris1974} is usually written with state~$1$
infected and state~$0$ healthy, but for consistency with KCSM notation we use the reversed labeling: state~$0$ is
infected and state~$1$ is healthy. With recovery rate one and infection
parameter~$\lambda\geq0$, its generator is
\begin{equation*}
\Lcal_{\mathrm{CP}}f(\eta)
=
\sum_{i\in\Lambda}
\bb{
1-\eta(i)
+
\lambda\eta(i)\sum_{j\in N(i)}\bb{1-\eta(j)}
}
\bb{f(\eta^i)-f(\eta)}.
\end{equation*}
Thus~$c_i^0(\vn)=1$, $c_i^1(\vn)=\lambda\abs{N(i)}$, and the only nonzero
nonconstant coefficients are~$c_i^1(\cb{j})=-\lambda$ for~$j\in N(i)$. The
coefficient criterion shows that the contact process is patch-positive with
patch threshold profile~$\mb p^\star=\mb 0$. Since the contact process is
monotone in configuration order, the classical monotone coupling gives
stronger comparisons than the centered-moment order considered here.
\end{remark}

\subsection{FA-1f}
\label{subsec:fa-1f-applications}

Fix~$p\in(0,1)$, write~$q=1-p$, and assume that~$N(i)\neq\vn$ for every
$i\in\Lambda$. In FA-1f~\cite{FredricksonAndersen1984,CancriniEtAl2008,
HartarskyToninelli2025}, state~$0$ is vacant and state~$1$ is occupied. A site
refreshes when at least one neighbour is vacant:
\begin{equation*}
\Lcal_{\mathrm{FA}}f(\eta)
=
\sum_{i\in\Lambda}
\bb{1-\chi_{N(i)}(\eta)}
\bb{E_{i,p}-I}f(\eta).
\end{equation*}
Its nonzero coefficients are
\begin{align*}
c_i^0(\vn)&=p,
&c_i^0\bb{N(i)}&=-p,
\\
c_i^1(\vn)&=q,
&c_i^1\bb{N(i)}&=-q.
\end{align*}
The Bernoulli product measure~$\mu_p$ is reversible, while the all-occupied
configuration is absorbing. For sufficiently large vacancy density,
Blondel et al.~\cite[Theorem~2.1 and Remarks~2.3--2.4]{BlondelEtAl2013}
proved convergence to~$\mu_p$ from initial laws having a uniform exponential
moment for the distance to the nearest vacancy. The rate is exponential on
one-dimensional polynomial-growth graphs and stretched exponential in higher
growth dimension. Related out-of-equilibrium results include
\cite{CancriniEtAl2010,MountfordValle2019,Mareche2019,
HartarskyToninelli2024}.

For~$\varepsilon\geq0$ and
$\mb s=(s_i)_{i\in\Lambda}\in[0,1]^\Lambda$, define
\begin{equation*}
\Lcal_{\mathrm{FA}}^{\varepsilon,\mb s}
=
\Lcal_{\mathrm{FA}}+\varepsilon\Ncal^{\mb s}.
\end{equation*}

\begin{proposition}[Patch positivity for FA-1f]
\label{prop:fa-1f-patch-positivity}
FA-1f is patch-positive with patch threshold profile~$\mb p^\star=\mb p$.
If~$\varepsilon>0$, then
$\Lcal_{\mathrm{FA}}^{\varepsilon,\mb s}$ is patch-positive if and only
if~$\mb s\geq\mb p$. In that case its patch threshold profile is again~$\mb p$.
\end{proposition}

\begin{proof}
For the only nonempty set with a nonzero coefficient,
\begin{equation*}
c_i^0\bb{N(i)}+c_i^1\bb{N(i)}=-1,
\qquad
c_i^1(\vn)c_i^0\bb{N(i)}
-c_i^0(\vn)c_i^1\bb{N(i)}=0.
\end{equation*}
Theorem~\ref{thm:patch-positivity-order} therefore gives patch positivity
and~$p_i^\star=p$. The added refresh changes the empty-neighbour coefficients
to
\begin{equation*}
c_i^0(\vn)=p+\varepsilon s_i,
\qquad
c_i^1(\vn)=q+\varepsilon\bb{1-s_i},
\end{equation*}
so the second coefficient expression becomes~$\varepsilon(s_i-p)$. The
nonempty coefficients, and hence the threshold profile, are unchanged.
\end{proof}

The equality case~$\mb s=\mb p$ is the usual soft FA-1f perturbation, up to a
deterministic rescaling of time. For the hard model, centered-moment comparison
gives a short transfer of known endpoint estimates. If
$\mu_u\preceq_*\nu\preceq_*\mu_v$ with~$v<1$, then convergence of the two
product endpoints implies convergence of~$\nu$ at the same rate, by
Corollary~\ref{cor:centered-moment-convergence}. This does not enlarge the
qualitative class treated in~\cite{BlondelEtAl2013}: ordinary monomial
comparison gives~$\nu(\chi_A)\leq v^{\abs A}$, and hence a uniform exponential
moment for the distance to the nearest vacancy. It does transfer any sharper
endpoint estimate available from model-specific arguments.

\subsection{Biased annihilating branching process}
\label{subsec:babp-applications}

Fix~$p\in(0,1)$, write~$q=1-p$, and assume that~$N(i)\neq\vn$ for every
$i\in\Lambda$. BABP~\cite{NeuhauserSudbury1993} uses the same refresh law as
FA-1f, with refresh rate equal to the number of vacant neighbours:
\begin{equation*}
\Lcal_{\mathrm{BABP}}f(\eta)
=
\sum_{i\in\Lambda}
\bb{\sum_{j\in N(i)}\bb{1-\eta(j)}}
\bb{E_{i,p}-I}f(\eta).
\end{equation*}
Here state~$0$ is vacant and state~$1$ is occupied. Its nonzero coefficients are
\begin{align*}
c_i^0(\vn)&=p\abs{N(i)},
&c_i^0\bb{\cb{j}}&=-p,
\\
c_i^1(\vn)&=q\abs{N(i)},
&c_i^1\bb{\cb{j}}&=-q,
\qquad j\in N(i).
\end{align*}
In the complementary particle variables~$\xi(i)=1-\eta(i)$, vacancies of
$\eta$ correspond to particles of~$\xi$, and~$q$ is their equilibrium density.
The product law~$\mu_p$ and
the particle-free state are invariant. In one dimension they generate all
stationary laws~\cite{NeuhauserSudbury1993,MartinelliShapiraToninelli2025}, and
the analogous statement holds among translation-invariant laws in higher
dimensions~\cite{MartinelliShapiraToninelli2025}. Sudbury proved qualitative
convergence from translation-invariant initial laws~\cite{Sudbury1997}.
On~$\Z^d$, every homogeneous product law~$\mu_u$ with~$u<1$ converges
exponentially to~$\mu_p$
\cite[Application~2 and Section~5.1.1]{MartinelliShapiraToninelli2025}. The
quasi-duality and thinning relations were introduced by Lloyd and
Sudbury~\cite{LloydSudbury1997}; the recent argument also covers inhomogeneous
product laws whose empty-site densities satisfy
$p_{0,i}\leq\sqrt p$ for every~$i$
\cite[Remark~5.5]{MartinelliShapiraToninelli2025}. Convergence from a finite
nonempty particle set remains open for part of the parameter range.

For~$\varepsilon\geq0$ and
$\mb s=(s_i)_{i\in\Lambda}\in[0,1]^\Lambda$, define
\begin{equation*}
\Lcal_{\mathrm{BABP}}^{\varepsilon,\mb s}
=
\Lcal_{\mathrm{BABP}}+\varepsilon\Ncal^{\mb s}.
\end{equation*}

\begin{proposition}[Patch positivity for BABP]
\label{prop:babp-patch-positivity}
BABP is patch-positive with patch threshold profile~$\mb p^\star=\mb p$.
If~$\varepsilon>0$, then
$\Lcal_{\mathrm{BABP}}^{\varepsilon,\mb s}$ is patch-positive if and only
if~$\mb s\geq\mb p$. In that case its patch threshold profile is again~$\mb p$.
\end{proposition}

\begin{proof}
For every~$j\in N(i)$,
\begin{equation*}
c_i^0\bb{\cb{j}}+c_i^1\bb{\cb{j}}=-1,
\qquad
c_i^1(\vn)c_i^0\bb{\cb{j}}
-c_i^0(\vn)c_i^1\bb{\cb{j}}=0.
\end{equation*}
Theorem~\ref{thm:patch-positivity-order} therefore gives patch positivity
and~$p_i^\star=p$. The added refresh changes the empty-neighbour coefficients
to
\begin{equation*}
c_i^0(\vn)=p\abs{N(i)}+\varepsilon s_i,
\qquad
c_i^1(\vn)=q\abs{N(i)}+\varepsilon\bb{1-s_i},
\end{equation*}
so the second coefficient expression becomes~$\varepsilon(s_i-p)$. The
nonempty coefficients, and hence the threshold profile, are unchanged.
\end{proof}

\begin{corollary}[Centered-moment comparison for BABP]
\label{cor:babp-centered-convergence}
Consider nearest-neighbour BABP on~$\Z^d$. Let~$0\leq u\leq v<1$ satisfy
$v\geq p$ and~$\abs{u-p}\leq v-p$. If a probability measure~$\nu$ satisfies
\begin{equation*}
(u-p)^{\abs A}
\leq
\nu\bb{\prod_{i\in A}\bb{\eta(i)-p}}
\leq
(v-p)^{\abs A}
\qquad
\text{for every }A\Subset\Z^d,
\end{equation*}
then~$\nu P_t$ converges exponentially to~$\mu_p$ on local functions.
\end{corollary}

\begin{proof}
The displayed inequalities are exactly
$\mu_u\preceq_*\nu\preceq_*\mu_v$. Both product endpoints converge
exponentially to~$\mu_p$ by
\cite[Application~2 and Section~5.1.1]{MartinelliShapiraToninelli2025}, so the
claim follows
from Corollary~\ref{cor:centered-moment-convergence}.
\end{proof}

The corollary includes dependent and non-translation-invariant laws, and the
lower comparison law may lie outside~$\mathcal M_*$. In particular, every
inhomogeneous product profile satisfying~$p\leq p_i\leq v<1$ converges
exponentially; this allows coordinates above the quasi-thinning threshold
$\sqrt p$. The comparison does not order finite-particle configurations between
nondegenerate product laws and does not resolve finite-seed convergence.

\subsection{Refresh noise}
\label{subsec:fa-babp-consequences}

Let~$\Lcal$ denote either~$\Lcal_{\mathrm{FA}}$ or~$\Lcal_{\mathrm{BABP}}$, and
write~$P_t^{\varepsilon,\mb s}$ for the semigroup generated by
$\Lcal+\varepsilon\Ncal^{\mb s}$.

\begin{proposition}[Noisy FA-1f and BABP]
\label{prop:fa-babp-consequences}
Assume that either~$\varepsilon=0$, or~$\varepsilon>0$ and~$\mb s\geq\mb p$.
\begin{enumerate}[label=\textup{(\roman*)}]
\item The centered-moment order about~$p$ is preserved:
\begin{equation*}
\mu\preceq_*\nu
\quad\Longrightarrow\quad
\mu P_t^{\varepsilon,\mb s}
\preceq_*
\nu P_t^{\varepsilon,\mb s}
\qquad
\text{for every }t\geq0.
\end{equation*}
In particular,~$\mathcal M_*$ is preserved.

\item Suppose that~$\varepsilon>0$, that~$\Lambda$ has polynomial growth of
exponent~$D$, and that
\begin{equation*}
\delta
=
\varepsilon\inf_{i\in\Lambda}\bb{1-s_i}
>
0.
\end{equation*}
Then the process has an invariant probability measure~$\pi$ and, for every
local function~$f$,
\begin{equation*}
\sup_{\mu\in\mathcal M_{-,K}}
\abs{(\mu P_t^{\varepsilon,\mb s})(f)-\pi(f)}
\leq
\bb{1+2K}K_f(1+t)^D e^{-\delta t/2}.
\end{equation*}
This holds for every~$K\geq0$ and~$t\geq0$. Here
$p^-=\max\cb{2p-1,0}$, so the covered initial laws include every~$\mu_u$ with
$u\geq p^-$ and every point mass on a configuration with finitely many sites
in state~$0$.
If~$p\leq\frac12$, then~$\pi$ is the unique invariant measure and
\begin{equation*}
\sup_{\eta\in\Omega}
\abs{P_t^{\varepsilon,\mb s}f(\eta)-\pi(f)}
\leq
K_f(1+t)^D e^{-\delta t/2}.
\end{equation*}
Its monomial moments are given by~\eqref{eq:full-patch-limit-moments}.
\end{enumerate}
\end{proposition}

\begin{proof}
The two patch-positivity propositions and
Theorem~\ref{thm:patch-positivity-order} give part~\textup{(i)}. For
part~\textup{(ii)}, the added refresh supplies independent~$1\to0$ transitions
at rate at least~$\delta$, and hence a pure-death component of rate~$\delta$.
Apply Theorem~\ref{thm:common-invariant-limit} and
Corollary~\ref{cor:uniform-ergodicity}.
\end{proof}
