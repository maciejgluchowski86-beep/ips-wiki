\subsubsection{Patch positivity and critical density}
\label{subsubsec:fa-1f-patches}

Write~$p=1-q$, set~$\mb p=(p)_{i\in\Lambda}$, and assume that~$N(i)\neq\vn$
for every~$i\in\Lambda$. Recall that FA-1f has generator
\begin{equation*}
\Lcal_{\mathrm{FA}}f(\eta)
=
\sum_{i\in\Lambda}
\bb{1-\chi_{N(i)}(\eta)}
\bb{E_{i,p}-I}f(\eta).
\end{equation*}
For~$\varepsilon\geq0$ and a refresh profile
$\mb s=(s_i)_{i\in\Lambda}\in[0,1]^\Lambda$, define
\begin{equation*}
\Lcal_{\mathrm{FA}}^{\varepsilon,\mb s}f
=
\Lcal_{\mathrm{FA}}f
+
\varepsilon
\sum_{i\in\Lambda}
\bb{E_{i,s_i}-I}f.
\end{equation*}

\begin{proposition}[Patch positivity for FA-1f]
\label{prop:fa-1f-patch-positivity}
FA-1f has patch positivity, and its patch critical profile is
$\mb p^{\star}=\mb p$. If~$\varepsilon>0$, then
$\Lcal_{\mathrm{FA}}^{\varepsilon,\mb s}$ has patch positivity if and only
if~$\mb s\geq\mb p$. Whenever this holds, its patch critical profile is
again~$\mb p$.
\end{proposition}

\begin{proof}
The only nonzero coefficients of the FA-1f rates are
\begin{align*}
c_i^0(\vn)&=p,
&c_i^0\bb{N(i)}&=-p,
\\
c_i^1(\vn)&=q,
&c_i^1\bb{N(i)}&=-q.
\end{align*}
For the only nonempty set with a nonzero coefficient, the two inequalities in
Theorem~\ref{thm:patch-positivity-criterion} reduce to
\begin{align*}
c_i^0\bb{N(i)}+c_i^1\bb{N(i)}
&=-1\leq0,
\\
c_i^1(\vn)c_i^0\bb{N(i)}
-c_i^0(\vn)c_i^1\bb{N(i)}
&=q(-p)-p(-q)=0.
\end{align*}
All other nonempty-neighbour coefficients vanish, proving patch positivity.
Proposition~\ref{prop:critical-profile-formula} gives
\begin{equation*}
p_i^{\star}
=
\frac{c_i^0\bb{N(i)}}
{c_i^0\bb{N(i)}+c_i^1\bb{N(i)}}
=
\frac{-p}{-p-q}
=
p.
\end{equation*}

The additional refresh changes only the empty-neighbour coefficients, replacing
them by
\begin{equation*}
c_i^0(\vn)=p+\varepsilon s_i,
\qquad
c_i^1(\vn)=q+\varepsilon\bb{1-s_i}.
\end{equation*}
The first coefficient inequality is unchanged, while the second becomes
\begin{equation*}
\bb{q+\varepsilon\bb{1-s_i}}(-p)
-
\bb{p+\varepsilon s_i}(-q)
=
\varepsilon\bb{s_i-p}.
\end{equation*}
Since~$\varepsilon>0$, the criterion holds exactly when~$s_i\geq p$ for every
site. The nonempty-neighbour coefficients are unchanged, so the critical-profile
formula still gives~$p_i^\star=p$.
\end{proof}

The equality case~$\mb s=\mb p$ is the usual soft FA-1f perturbation. It adds
unconstrained refreshes with the reversible Bernoulli law of FA-1f and therefore has
a positive spectral gap. We do not use these standard facts below.

\subsubsection{Patch positivity and critical density}
\label{subsubsec:babp-patches}

Write~$p=1-q$, set~$\mb p=(p)_{i\in\Lambda}$, and assume that~$N(i)\neq\vn$
for every~$i\in\Lambda$. Recall that BABP has generator
\begin{equation*}
\Lcal_{\mathrm{BABP}}f(\eta)
=
\sum_{i\in\Lambda}
\bb{\sum_{j\in N(i)}\bb{1-\eta(j)}}
\bb{E_{i,p}-I}f(\eta).
\end{equation*}
For~$\varepsilon\geq0$ and a refresh profile
$\mb s=(s_i)_{i\in\Lambda}\in[0,1]^\Lambda$, define
\begin{equation*}
\Lcal_{\mathrm{BABP}}^{\varepsilon,\mb s}f
=
\Lcal_{\mathrm{BABP}}f
+
\varepsilon
\sum_{i\in\Lambda}
\bb{E_{i,s_i}-I}f.
\end{equation*}

\begin{proposition}[Patch positivity for BABP]
\label{prop:babp-patch-positivity}
BABP has patch positivity, and its patch critical profile is
$\mb p^\star=\mb p$. If~$\varepsilon>0$, then
$\Lcal_{\mathrm{BABP}}^{\varepsilon,\mb s}$ has patch positivity if and only
if~$\mb s\geq\mb p$. Whenever this holds, its patch critical profile is
again~$\mb p$.
\end{proposition}

\begin{proof}
The only nonzero coefficients of the BABP rates are
\begin{align*}
c_i^0(\vn)&=p\abs{N(i)},
&c_i^0\bb{\cb{j}}&=-p,
\\
c_i^1(\vn)&=q\abs{N(i)},
&c_i^1\bb{\cb{j}}&=-q,
\qquad j\in N(i).
\end{align*}
For every~$j\in N(i)$, the two inequalities in
Theorem~\ref{thm:patch-positivity-criterion} reduce to
\begin{align*}
c_i^0\bb{\cb{j}}+c_i^1\bb{\cb{j}}
&=-1\leq0,
\\
c_i^1(\vn)c_i^0\bb{\cb{j}}
-c_i^0(\vn)c_i^1\bb{\cb{j}}
&=q\abs{N(i)}(-p)-p\abs{N(i)}(-q)=0.
\end{align*}
All other nonempty-neighbour coefficients vanish, proving patch positivity.
Proposition~\ref{prop:critical-profile-formula} gives
\begin{equation*}
p_i^\star
=
\frac{c_i^0\bb{\cb{j}}}
{c_i^0\bb{\cb{j}}+c_i^1\bb{\cb{j}}}
=
\frac{-p}{-p-q}
=
p.
\end{equation*}

The additional refresh changes only the empty-neighbour coefficients, replacing
them by
\begin{align*}
c_i^0(\vn)&=p\abs{N(i)}+\varepsilon s_i,
&
c_i^1(\vn)&=q\abs{N(i)}+\varepsilon\bb{1-s_i}.
\end{align*}
The first coefficient inequality is unchanged, while the second becomes
\begin{equation*}
\bb{q\abs{N(i)}+\varepsilon\bb{1-s_i}}(-p)
-
\bb{p\abs{N(i)}+\varepsilon s_i}(-q)
=
\varepsilon\bb{s_i-p}.
\end{equation*}
Since~$\varepsilon>0$, the criterion holds exactly when~$s_i\geq p$ for every
site. The nonempty-neighbour coefficients are unchanged, so the critical-profile
formula still gives~$p_i^\star=p$.
\end{proof}

\subsection{Consequences}
\label{subsec:fa-babp-consequences}

Let~$\Lcal$ denote either~$\Lcal_{\mathrm{FA}}$ or~$\Lcal_{\mathrm{BABP}}$.
For~$\varepsilon\geq0$ and~$\mb s=(s_i)_{i\in\Lambda}\in[0,1]^\Lambda$, set
\begin{equation*}
\Lcal^{\varepsilon,\mb s}f
=
\Lcal f
+
\varepsilon\sum_{i\in\Lambda}\bb{E_{i,s_i}-I}f,
\end{equation*}
and write~$P_t^{\varepsilon,\mb s}$ for its semigroup. When~$\varepsilon=0$,
the profile~$\mb s$ is irrelevant. For either model, the class from
Section~\ref{sec:moment-order} is
\begin{equation*}
\mathcal M_*
=
\cb{
 \mu:
 \mu\bb{\prod_{i\in A}\bb{\eta(i)-p}}\geq0
 \text{ for every }A\Subset\Lambda
}.
\end{equation*}
The order~$\preceq_*$ compares the same centered moments. For~$u\in[0,1]$,
write~$\mu_u$ for the homogeneous Bernoulli product measure of one-density~$u$.

\begin{proposition}[Consequences for FA-1f and BABP]
\label{prop:fa-babp-consequences}
Assume that either~$\varepsilon=0$, or~$\varepsilon>0$ and~$\mb s\geq\mb p$.
\begin{enumerate}[label=\textup{(\roman*)}]
\item The centered-moment order is preserved:
\begin{equation*}
\mu\preceq_*\nu
\quad\Longrightarrow\quad
\mu P_t^{\varepsilon,\mb s}
\preceq_*
\nu P_t^{\varepsilon,\mb s}
\qquad
\text{for every }t\geq0.
\end{equation*}
In particular,~$\mathcal M_*$ is preserved and, if~$p\leq u\leq v\leq1$, then
\begin{equation*}
(\mu_u P_t^{\varepsilon,\mb s})(\chi_A)
\leq
(\mu_v P_t^{\varepsilon,\mb s})(\chi_A)
\qquad
\text{for every }A\Subset\Lambda\text{ and }t\geq0.
\end{equation*}
Moreover, if
$\underline\mu\preceq_*\mu\preceq_*\overline\mu$ and both
$\underline\mu P_t^{\varepsilon,\mb s}$ and
$\overline\mu P_t^{\varepsilon,\mb s}$ converge weakly to the same probability
measure~$\pi$, then~$\mu P_t^{\varepsilon,\mb s}\Rightarrow\pi$.

\item Suppose that~$\varepsilon>0$, that~$\Lambda$ has polynomial growth of
exponent~$D$, and that
\begin{equation*}
\delta
=
\varepsilon\inf_{i\in\Lambda}\bb{1-s_i}
>
0.
\end{equation*}
Then there is an invariant probability measure~$\pi$ such that, for every local
function~$f$, there is~$K_f<\infty$ satisfying
\begin{equation*}
\sup_{\mu\in\mathcal M_{-,K}}
\abs{(\mu P_t^{\varepsilon,\mb s})(f)-\pi(f)}
\leq
\bb{1+2K}K_f(1+t)^D e^{-\delta t/2}
\end{equation*}
for every~$K\geq0$ and~$t\geq0$. Consequently,
$\mu P_t^{\varepsilon,\mb s}\Rightarrow\pi$ for every~$\mu\in\mathcal M_-$.
Here
\begin{equation*}
p^-
=
\max\cb{2p-1,0},
\end{equation*}
so this includes every~$\mu_u$ with~$u\geq p^-$ and every point mass on a
configuration with finitely many zeroes. If~$p\leq\frac12$, then~$\pi$ is the
unique invariant measure and
\begin{equation*}
\sup_{\eta\in\Omega}
\abs{P_t^{\varepsilon,\mb s}f(\eta)-\pi(f)}
\leq
K_f(1+t)^D e^{-\delta t/2}.
\end{equation*}
The monomial moments of~$\pi$ are given by
\eqref{eq:full-patch-limit-moments}.
\end{enumerate}
\end{proposition}

\begin{proof}
Propositions~\ref{prop:fa-1f-patch-positivity} and
\ref{prop:babp-patch-positivity} give patch positivity and critical profile
$\mb p^\star=\mb p$ in every case covered by the statement. Part~\textup{(i)}
then follows from Theorem~\ref{thm:centered-moment-order} and
Corollary~\ref{cor:product-profile-comparisons}. For the final assertion in that
part, order preservation squeezes every centered monomial between its values
under the two extremal evolutions. Centered monomials span the local functions,
so the weak convergence follows.

For part~\textup{(ii)}, the added refresh supplies an environment-independent
$1\to0$ rate~$\varepsilon\bb{1-s_i}\geq\delta$ at every site. Theorem~\ref{thm:common-invariant-limit} and Corollary~\ref{cor:uniform-ergodicity}
therefore give the stated conclusions. The formula for~$p^-$ and the listed
initial laws follow from the description of~$\mathcal M_-$ in
Section~\ref{subsec:convergence-statement}.
\end{proof}

The same conclusions hold for the two different facilitating mechanisms:
FA-1f records whether a vacant neighbour is present, whereas BABP counts the
vacant neighbours.
