We now apply our main results to several familiar spin systems. FA-1f is a
basic KCSM for which patch positivity holds, while higher-order FA models show
that patch positivity does not hold for all KCSMs. BABP gives an example
outside the KCSM class, and the contact process gives an example which is also
monotone in the usual configuration order. We then use FA-1f and BABP to
illustrate the centered-moment comparison and the convergence results proved
above.

\subsection{FA-1f}
\label{subsec:fa-1f-applications}

Fix~$p\in(0,1)$ and assume that~$N(i)\neq\vn$ for every~$i\in\Lambda$, so there are no isolated sites.
In FA-1f~\cite{FredricksonAndersen1984,CancriniEtAl2008,HartarskyToninelli2025},
state~$0$ is vacant and state~$1$ is occupied. A site is refreshed from the
Bernoulli law with parameter~$p$ whenever at least one neighbour is vacant, so
\begin{equation*}
\lambda_i(\eta)=1-\chi_{N(i)}(\eta),
\qquad
\rho_i(\eta)=p.
\end{equation*}
The only nonzero nonconstant M\"obius coefficient of~$\lambda_i$ is
\begin{equation*}
\widehat{\lambda_i}\bb{N(i)}=-1,
\end{equation*}
while~$\lambda_i\bb{\rho_i-\rho_i(\mb 0)}=0$. Hence
Theorem~\ref{thm:patch-positivity-order} shows that FA-1f is patch-positive.
Moreover,
\begin{equation*}
\frac{\widehat{\lambda_i\rho_i}\bb{N(i)}}
{\widehat{\lambda_i}\bb{N(i)}}=p,
\end{equation*}
so its patch threshold profile is~$\mb p^\star=\mb p$.

\begin{remark}
Patch positivity does not hold for all KCSMs with monotone constraints. For
example, for cooperative FA-$j$f, i.e. with~$j\geq2$, the refresh rate is the indicator that at
least~$j$ neighbours are vacant. Although this constraint is monotone with
respect to vacancies, some of its M\"obius coefficients are positive, so the
first condition in~\eqref{eq:principal-coefficient-criterion} fails.
\end{remark}

\subsection{Biased annihilating branching process}
\label{subsec:babp-applications}

Fix~$p\in(0,1)$ and write~$q=1-p$. The biased annihilating branching process
(BABP)~\cite{NeuhauserSudbury1993} has the same Bernoulli refresh law as
FA-1f, but the refresh rate is the number of vacant neighbours:
\begin{equation*}
\lambda_i(\eta)
=
\sum_{j\in N(i)}\bb{1-\eta(j)},
\qquad
\rho_i(\eta)=p.
\end{equation*}
In the complementary variables~$\xi(i)=1-\eta(i)$, vacancies correspond to
particles and~$q$ is their equilibrium density. The only nonzero nonconstant
M\"obius coefficients of~$\lambda_i$ are
\begin{equation*}
\widehat{\lambda_i}\bb{\cb{j}}=-1,
\qquad j\in N(i),
\end{equation*}
while again~$\lambda_i\bb{\rho_i-\rho_i(\mb 0)}=0$. Hence
Theorem~\ref{thm:patch-positivity-order} shows that BABP is patch-positive.
Moreover, for every~$j\in N(i)$,
\begin{equation*}
\frac{\widehat{\lambda_i\rho_i}\bb{\cb{j}}}
{\widehat{\lambda_i}\bb{\cb{j}}}=p,
\end{equation*}
so its patch threshold profile is~$\mb p^\star=\mb p$.

On~$\Z^d$, every homogeneous product law~$\mu_u$ with~$u<1$ converges
exponentially to~$\mu_p$
\cite[Application~2 and Section~5.1.1]{MartinelliShapiraToninelli2025}.
The same argument also treats inhomogeneous product laws whose one-site
probabilities~$p_i$ of state~$1$ satisfy
\begin{equation*}
p_i\leq\sqrt p
\qquad\text{for every }i
\end{equation*}
\cite[Remark~5.5]{MartinelliShapiraToninelli2025}. The centered-moment order
allows us to go beyond this restriction and beyond product initial laws.

\begin{corollary}[Centered-moment comparison for BABP]
\label{cor:babp-centered-convergence}
Consider nearest-neighbour BABP on~$\Z^d$. Let~$0\leq u\leq v<1$ satisfy
$v\geq p$ and~$\abs{u-p}\leq v-p$. If a probability measure~$\nu$ satisfies
\begin{equation*}
(u-p)^{\abs A}
\leq
\nu\bb{\prod_{i\in A}\bb{\eta(i)-p}}
\leq
(v-p)^{\abs A}
\qquad
\text{for every }A\Subset\Z^d,
\end{equation*}
then~$\nu P_t$ converges exponentially to~$\mu_p$ on local functions.
\end{corollary}

\begin{proof}
The inequalities above are equivalent to
$\mu_u\preceq_*\nu\preceq_*\mu_v$. Both product endpoints converge
exponentially to~$\mu_p$ by
\cite[Application~2 and Section~5.1.1]{MartinelliShapiraToninelli2025}, so the
claim follows from Corollary~\ref{cor:centered-moment-convergence}.
\end{proof}

This includes dependent and non-translation-invariant initial laws. In
particular, taking~$u=p$, every inhomogeneous product law with one-site
parameters~$p\leq p_i\leq v<1$ satisfies the assumptions and therefore
converges exponentially to~$\mu_p$. Thus the coordinates may lie above the
previous bound~$\sqrt p$, provided they are uniformly bounded away from~$1$.
The comparison does not place finite-particle configurations between
nondegenerate product laws and hence does not address finite-seed convergence.
An analogous centered-moment sandwiching argument applies to FA-1f, but does
not enlarge the qualitative class of initial laws covered by the known FA-1f
convergence results~\cite{BlondelEtAl2013}.

\subsection{Contact process}
\label{subsec:contact-benchmark}

The contact process~\cite{Harris1974} is usually written with state~$1$
infected and state~$0$ healthy. We use the reversed labeling, so that state~$0$
is infected and state~$1$ is healthy. With recovery rate one and infection
parameter~$\beta\geq0$, its refresh representation is
\begin{equation*}
\lambda_i(\eta)
=
1+\beta\sum_{j\in N(i)}\bb{1-\eta(j)},
\qquad
\rho_i(\eta)
=
\frac{1}{1+\beta\sum_{j\in N(i)}\bb{1-\eta(j)}}.
\end{equation*}
Thus~$\lambda_i\rho_i=1$, and the only nonzero nonconstant M\"obius
coefficients of~$\lambda_i$ are
\begin{equation*}
\widehat{\lambda_i}\bb{\cb{j}}=-\beta,
\qquad j\in N(i).
\end{equation*}
For every nonempty~$S\subseteq N(i)$,
\begin{equation*}
\widehat{\lambda_i\bb{\rho_i-\rho_i(\mb0)}}(S)
=
-\rho_i(\mb0)\widehat{\lambda_i}(S)
\geq0.
\end{equation*}
Hence the contact process is patch-positive. Since~$\lambda_i\rho_i$ is
constant, its patch threshold profile is~$\mb p^\star=\mb0$. The contact
process is also monotone in the usual configuration order, for which the
classical monotone coupling gives stronger comparisons than the centered-moment
order considered here.

\subsection{Perturbed FA-1f and BABP}
\label{subsec:fa-babp-consequences}

The FA-1f and BABP dynamics above can stop updating at a site when all of its
neighbours are occupied. We now add an unconstrained Bernoulli refresh at every
site. When the added refresh puts enough mass on state~$1$, patch positivity is
preserved. If it also puts state~$0$ with a uniformly positive probability, the
perturbed process has the pure-death component needed for the convergence
results of Section~\ref{sec:main-results}. The same perturbation also gives a
comparison between the limiting law and the Bernoulli product measure~$\mu_p$.

Let~$\Lcal$ denote either FA-1f or BABP with refresh parameter~$p$. Assume that
$N(i)\neq\vn$ for every~$i\in\Lambda$. For~$\varepsilon>0$ and
$\mb s=(s_i)_{i\in\Lambda}\in[0,1]^\Lambda$, set
\begin{equation*}
\Lcal^{\varepsilon,\mb s}
=
\Lcal+\varepsilon\Ncal^{\mb s},
\end{equation*}
and write~$P_t^{\varepsilon,\mb s}$ for its semigroup.

\begin{proposition}[Perturbed FA-1f and BABP]
\label{prop:fa-babp-consequences}
\leavevmode
\begin{enumerate}[label=\textup{(\roman*)}]
\item The perturbed system is patch-positive if and only if
$\mb s\geq\mb p$. In that case its patch threshold profile is
$\mb p^\star=\mb p$, and the centered-moment order about~$p$ is preserved by
$P_t^{\varepsilon,\mb s}$.

\item Suppose that~$\mb s\geq\mb p$, that~$\Lambda$ has polynomial growth of
exponent~$D$, and that
\begin{equation*}
\delta
=
\varepsilon\inf_{i\in\Lambda}\bb{1-s_i}
>
0.
\end{equation*}
Then there is an invariant probability measure~$\pi$ such that, for every
local function~$f$, there is~$K_f<\infty$ satisfying
\begin{equation*}
\sup_{\mu\in\mathcal M_{-,K}}
\abs{(\mu P_t^{\varepsilon,\mb s})(f)-\pi(f)}
\leq
\bb{1+2K}K_f(1+t)^D e^{-\delta t/2}
\end{equation*}
for every~$K\geq0$ and~$t\geq0$. Consequently,
$\mu P_t^{\varepsilon,\mb s}\Rightarrow\pi$ for every~$\mu\in\mathcal M_-$.
In particular, this includes every homogeneous product law~$\mu_u$ with
$u\geq p^-:=\max\cb{2p-1,0}$ and every point mass on a configuration with
finitely many sites in state~$0$.

\item Under the assumptions of part~\textup{(ii)},
\begin{equation*}
p^{\abs A}
\leq
\pi(\chi_A)
\qquad
\text{for every }A\Subset\Lambda.
\end{equation*}
If~$\mb s=\mb p$, then~$\pi=\mu_p$.

\item Under the assumptions of part~\textup{(ii)}, if~$p\leq\frac12$, then
$\pi$ is the unique invariant measure and
\begin{equation*}
\sup_{\eta\in\Omega}
\abs{P_t^{\varepsilon,\mb s}f(\eta)-\pi(f)}
\leq
K_f(1+t)^D e^{-\delta t/2}
\end{equation*}
for every local function~$f$ and~$t\geq0$.
\end{enumerate}
\end{proposition}

\begin{proof}
At site~$i$, the perturbed dynamics have refresh rate and refresh parameter
\begin{equation*}
\lambda_i^\varepsilon(\eta)=\lambda_i(\eta)+\varepsilon,
\qquad
\rho_i^\varepsilon(\eta)
=
\frac{p\lambda_i(\eta)+\varepsilon s_i}
{\lambda_i(\eta)+\varepsilon}.
\end{equation*}
Thus
\begin{equation*}
\lambda_i^\varepsilon\rho_i^\varepsilon
=
p\lambda_i+\varepsilon s_i.
\end{equation*}
For every nonempty~$S\subseteq N(i)$,
\begin{equation*}
\widehat{\lambda_i^\varepsilon}(S)=\widehat{\lambda_i}(S),
\qquad
\widehat{\lambda_i^\varepsilon\rho_i^\varepsilon}(S)
=
p\widehat{\lambda_i}(S),
\end{equation*}
and hence
\begin{equation*}
\widehat{\lambda_i^\varepsilon
\bb{\rho_i^\varepsilon-\rho_i^\varepsilon(\mb0)}}(S)
=
\bb{p-\rho_i^\varepsilon(\mb0)}\widehat{\lambda_i}(S).
\end{equation*}
The nonconstant coefficients of~$\lambda_i$ are nonpositive in both models,
and at each site at least one is strictly negative. Therefore
Theorem~\ref{thm:patch-positivity-order} shows that the perturbed system is
patch-positive exactly when~$\rho_i^\varepsilon(\mb0)\geq p$ for every~$i$.
Since
\begin{equation*}
\rho_i^\varepsilon(\mb0)
=
\frac{p\lambda_i(\mb0)+\varepsilon s_i}
{\lambda_i(\mb0)+\varepsilon},
\end{equation*}
this is equivalent to~$s_i\geq p$. Whenever
$\widehat{\lambda_i}(S)<0$,
\begin{equation*}
\frac{\widehat{\lambda_i^\varepsilon\rho_i^\varepsilon}(S)}
{\widehat{\lambda_i^\varepsilon}(S)}
=p,
\end{equation*}
so the patch threshold remains~$p$ at every site. The centered-moment
comparison in part~\textup{(i)} now follows from the same theorem.

For part~\textup{(ii)}, the added refresh at site~$i$ contains pure deaths at
rate~$\varepsilon(1-s_i)$. Thus~$\delta>0$ gives a uniform pure-death
component of rate~$\delta$, and
Theorem~\ref{thm:common-invariant-limit} gives the stated invariant measure,
estimate, and convergence from~$\mathcal M_-$. The two classes of initial laws
listed in the statement are contained in~$\mathcal M_-$ by the discussion in
Section~\ref{sec:main-results}.

For part~\textup{(iii)}, set
\begin{equation*}
d_i=\varepsilon\frac{s_i-p}{p}\geq0
\end{equation*}
and define
\begin{equation*}
\widetilde\Lcal
=
\Lcal^{\varepsilon,\mb s}
+
\sum_{i\in\Lambda}d_i\bb{E_{i,0}-I}.
\end{equation*}
A direct calculation gives
\begin{equation*}
\widetilde\Lcal
=
\Lcal
+
\sum_{i\in\Lambda}
\frac{\varepsilon s_i}{p}\bb{E_{i,p}-I}.
\end{equation*}
The added refreshes on the right have parameter~$p$, so the coefficient
calculation from part~\textup{(i)} shows that~$\widetilde\Lcal$ is
patch-positive with threshold~$p$. Moreover,~$\mu_p$ is invariant for
$\widetilde\Lcal$. If~$\widetilde P_t$ denotes its semigroup,
Corollary~\ref{cor:pure-death-comparison}, applied with initial law~$\mu_p$,
gives
\begin{equation*}
p^{\abs A}
=
(\mu_p\widetilde P_t)(\chi_A)
\leq
(\mu_pP_t^{\varepsilon,\mb s})(\chi_A).
\end{equation*}
Since~$\mu_p\in\mathcal M_*$, part~\textup{(ii)} allows us to let~$t\to\infty$
and obtain~$p^{\abs A}\leq\pi(\chi_A)$. If~$\mb s=\mb p$, then the original
perturbed dynamics themselves preserve~$\mu_p$, so convergence from~$\mu_p$
identifies~$\pi$ with~$\mu_p$.

Finally, when~$p\leq\frac12$, the threshold profile satisfies
$\mb p^\star\leq\mbs{\tfrac12}$. Part~\textup{(iv)} therefore follows from
Corollary~\ref{cor:uniform-ergodicity}.
\end{proof}

For FA-1f, the choice~$\mb s=\mb p$ is the usual soft FA-1f perturbation, up
to a deterministic rescaling of time.

\begin{remark}[Contact process]
\label{subsec:contact-benchmark}

The contact process~\cite{Harris1974} is usually written with state~$1$
infected and state~$0$ healthy, but for consistency with KCSM notation we use the reversed labeling: state~$0$ is
infected and state~$1$ is healthy. With recovery rate one and infection
parameter~$\lambda\geq0$, its generator is
\begin{equation*}
\Lcal_{\mathrm{CP}}f(\eta)
=
\sum_{i\in\Lambda}
\bb{
1-\eta(i)
+
\lambda\eta(i)\sum_{j\in N(i)}\bb{1-\eta(j)}
}
\bb{f(\eta^i)-f(\eta)}.
\end{equation*}
Thus~$c_i^0(\vn)=1$, $c_i^1(\vn)=\lambda\abs{N(i)}$, and the only nonzero
nonconstant coefficients are~$c_i^1(\cb{j})=-\lambda$ for~$j\in N(i)$. The
coefficient criterion shows that the contact process is patch-positive with
patch threshold profile~$\mb p^\star=\mb 0$. Since the contact process is
monotone in configuration order, the classical monotone coupling gives
stronger comparisons than the centered-moment order considered here.
\end{remark}

\subsection{FA-1f}
\label{subsec:fa-1f-applications}

Fix~$p\in(0,1)$, write~$q=1-p$, and assume that~$N(i)\neq\vn$ for every
$i\in\Lambda$. In FA-1f~\cite{FredricksonAndersen1984,CancriniEtAl2008,
HartarskyToninelli2025}, state~$0$ is vacant and state~$1$ is occupied. A site
refreshes when at least one neighbour is vacant:
\begin{equation*}
\Lcal_{\mathrm{FA}}f(\eta)
=
\sum_{i\in\Lambda}
\bb{1-\chi_{N(i)}(\eta)}
\bb{E_{i,p}-I}f(\eta).
\end{equation*}
Its nonzero coefficients are
\begin{align*}
c_i^0(\vn)&=p,
&c_i^0\bb{N(i)}&=-p,
\\
c_i^1(\vn)&=q,
&c_i^1\bb{N(i)}&=-q.
\end{align*}
The Bernoulli product measure~$\mu_p$ is reversible, while the all-occupied
configuration is absorbing. For sufficiently large vacancy density,
Blondel et al.~\cite[Theorem~2.1 and Remarks~2.3--2.4]{BlondelEtAl2013}
proved convergence to~$\mu_p$ from initial laws having a uniform exponential
moment for the distance to the nearest vacancy. The rate is exponential on
one-dimensional polynomial-growth graphs and stretched exponential in higher
growth dimension. Related out-of-equilibrium results include
\cite{CancriniEtAl2010,MountfordValle2019,Mareche2019,
HartarskyToninelli2024}.

For~$\varepsilon\geq0$ and
$\mb s=(s_i)_{i\in\Lambda}\in[0,1]^\Lambda$, define
\begin{equation*}
\Lcal_{\mathrm{FA}}^{\varepsilon,\mb s}
=
\Lcal_{\mathrm{FA}}+\varepsilon\Ncal^{\mb s}.
\end{equation*}

\begin{proposition}[Patch positivity for FA-1f]
\label{prop:fa-1f-patch-positivity}
FA-1f is patch-positive with patch threshold profile~$\mb p^\star=\mb p$.
If~$\varepsilon>0$, then
$\Lcal_{\mathrm{FA}}^{\varepsilon,\mb s}$ is patch-positive if and only
if~$\mb s\geq\mb p$. In that case its patch threshold profile is again~$\mb p$.
\end{proposition}

\begin{proof}
For the only nonempty set with a nonzero coefficient,
\begin{equation*}
c_i^0\bb{N(i)}+c_i^1\bb{N(i)}=-1,
\qquad
c_i^1(\vn)c_i^0\bb{N(i)}
-c_i^0(\vn)c_i^1\bb{N(i)}=0.
\end{equation*}
Theorem~\ref{thm:patch-positivity-order} therefore gives patch positivity
and~$p_i^\star=p$. The added refresh changes the empty-neighbour coefficients
to
\begin{equation*}
c_i^0(\vn)=p+\varepsilon s_i,
\qquad
c_i^1(\vn)=q+\varepsilon\bb{1-s_i},
\end{equation*}
so the second coefficient expression becomes~$\varepsilon(s_i-p)$. The
nonempty coefficients, and hence the threshold profile, are unchanged.
\end{proof}

The equality case~$\mb s=\mb p$ is the usual soft FA-1f perturbation, up to a
deterministic rescaling of time. For the hard model, centered-moment comparison
gives a short transfer of known endpoint estimates. If
$\mu_u\preceq_*\nu\preceq_*\mu_v$ with~$v<1$, then convergence of the two
product endpoints implies convergence of~$\nu$ at the same rate, by
Corollary~\ref{cor:centered-moment-convergence}. This does not enlarge the
qualitative class treated in~\cite{BlondelEtAl2013}: ordinary monomial
comparison gives~$\nu(\chi_A)\leq v^{\abs A}$, and hence a uniform exponential
moment for the distance to the nearest vacancy. It does transfer any sharper
endpoint estimate available from model-specific arguments.

\subsection{Biased annihilating branching process}
\label{subsec:babp-applications}

Fix~$p\in(0,1)$, write~$q=1-p$, and assume that~$N(i)\neq\vn$ for every
$i\in\Lambda$. BABP~\cite{NeuhauserSudbury1993} uses the same refresh law as
FA-1f, with refresh rate equal to the number of vacant neighbours:
\begin{equation*}
\Lcal_{\mathrm{BABP}}f(\eta)
=
\sum_{i\in\Lambda}
\bb{\sum_{j\in N(i)}\bb{1-\eta(j)}}
\bb{E_{i,p}-I}f(\eta).
\end{equation*}
Here state~$0$ is vacant and state~$1$ is occupied. Its nonzero coefficients are
\begin{align*}
c_i^0(\vn)&=p\abs{N(i)},
&c_i^0\bb{\cb{j}}&=-p,
\\
c_i^1(\vn)&=q\abs{N(i)},
&c_i^1\bb{\cb{j}}&=-q,
\qquad j\in N(i).
\end{align*}
In the complementary particle variables~$\xi(i)=1-\eta(i)$, vacancies of
$\eta$ correspond to particles of~$\xi$, and~$q$ is their equilibrium density.
The product law~$\mu_p$ and
the particle-free state are invariant. In one dimension they generate all
stationary laws~\cite{NeuhauserSudbury1993,MartinelliShapiraToninelli2025}, and
the analogous statement holds among translation-invariant laws in higher
dimensions~\cite{MartinelliShapiraToninelli2025}. Sudbury proved qualitative
convergence from translation-invariant initial laws~\cite{Sudbury1997}.
On~$\Z^d$, every homogeneous product law~$\mu_u$ with~$u<1$ converges
exponentially to~$\mu_p$
\cite[Application~2 and Section~5.1.1]{MartinelliShapiraToninelli2025}. The
quasi-duality and thinning relations were introduced by Lloyd and
Sudbury~\cite{LloydSudbury1997}; the recent argument also covers inhomogeneous
product laws whose empty-site densities satisfy
$p_{0,i}\leq\sqrt p$ for every~$i$
\cite[Remark~5.5]{MartinelliShapiraToninelli2025}. Convergence from a finite
nonempty particle set remains open for part of the parameter range.

For~$\varepsilon\geq0$ and
$\mb s=(s_i)_{i\in\Lambda}\in[0,1]^\Lambda$, define
\begin{equation*}
\Lcal_{\mathrm{BABP}}^{\varepsilon,\mb s}
=
\Lcal_{\mathrm{BABP}}+\varepsilon\Ncal^{\mb s}.
\end{equation*}

\begin{proposition}[Patch positivity for BABP]
\label{prop:babp-patch-positivity}
BABP is patch-positive with patch threshold profile~$\mb p^\star=\mb p$.
If~$\varepsilon>0$, then
$\Lcal_{\mathrm{BABP}}^{\varepsilon,\mb s}$ is patch-positive if and only
if~$\mb s\geq\mb p$. In that case its patch threshold profile is again~$\mb p$.
\end{proposition}

\begin{proof}
For every~$j\in N(i)$,
\begin{equation*}
c_i^0\bb{\cb{j}}+c_i^1\bb{\cb{j}}=-1,
\qquad
c_i^1(\vn)c_i^0\bb{\cb{j}}
-c_i^0(\vn)c_i^1\bb{\cb{j}}=0.
\end{equation*}
Theorem~\ref{thm:patch-positivity-order} therefore gives patch positivity
and~$p_i^\star=p$. The added refresh changes the empty-neighbour coefficients
to
\begin{equation*}
c_i^0(\vn)=p\abs{N(i)}+\varepsilon s_i,
\qquad
c_i^1(\vn)=q\abs{N(i)}+\varepsilon\bb{1-s_i},
\end{equation*}
so the second coefficient expression becomes~$\varepsilon(s_i-p)$. The
nonempty coefficients, and hence the threshold profile, are unchanged.
\end{proof}

\begin{corollary}[Centered-moment comparison for BABP]
\label{cor:babp-centered-convergence}
Consider nearest-neighbour BABP on~$\Z^d$. Let~$0\leq u\leq v<1$ satisfy
$v\geq p$ and~$\abs{u-p}\leq v-p$. If a probability measure~$\nu$ satisfies
\begin{equation*}
(u-p)^{\abs A}
\leq
\nu\bb{\prod_{i\in A}\bb{\eta(i)-p}}
\leq
(v-p)^{\abs A}
\qquad
\text{for every }A\Subset\Z^d,
\end{equation*}
then~$\nu P_t$ converges exponentially to~$\mu_p$ on local functions.
\end{corollary}

\begin{proof}
The displayed inequalities are exactly
$\mu_u\preceq_*\nu\preceq_*\mu_v$. Both product endpoints converge
exponentially to~$\mu_p$ by
\cite[Application~2 and Section~5.1.1]{MartinelliShapiraToninelli2025}, so the
claim follows
from Corollary~\ref{cor:centered-moment-convergence}.
\end{proof}

The corollary includes dependent and non-translation-invariant laws, and the
lower comparison law may lie outside~$\mathcal M_*$. In particular, every
inhomogeneous product profile satisfying~$p\leq p_i\leq v<1$ converges
exponentially; this allows coordinates above the quasi-thinning threshold
$\sqrt p$. The comparison does not order finite-particle configurations between
nondegenerate product laws and does not resolve finite-seed convergence.

\subsection{Refresh noise}
\label{subsec:fa-babp-consequences}

Let~$\Lcal$ denote either~$\Lcal_{\mathrm{FA}}$ or~$\Lcal_{\mathrm{BABP}}$, and
write~$P_t^{\varepsilon,\mb s}$ for the semigroup generated by
$\Lcal+\varepsilon\Ncal^{\mb s}$.

\begin{proposition}[Noisy FA-1f and BABP]
\label{prop:fa-babp-consequences}
Assume that either~$\varepsilon=0$, or~$\varepsilon>0$ and~$\mb s\geq\mb p$.
\begin{enumerate}[label=\textup{(\roman*)}]
\item The centered-moment order about~$p$ is preserved:
\begin{equation*}
\mu\preceq_*\nu
\quad\Longrightarrow\quad
\mu P_t^{\varepsilon,\mb s}
\preceq_*
\nu P_t^{\varepsilon,\mb s}
\qquad
\text{for every }t\geq0.
\end{equation*}
In particular,~$\mathcal M_*$ is preserved.

\item Suppose that~$\varepsilon>0$, that~$\Lambda$ has polynomial growth of
exponent~$D$, and that
\begin{equation*}
\delta
=
\varepsilon\inf_{i\in\Lambda}\bb{1-s_i}
>
0.
\end{equation*}
Then the process has an invariant probability measure~$\pi$ and, for every
local function~$f$,
\begin{equation*}
\sup_{\mu\in\mathcal M_{-,K}}
\abs{(\mu P_t^{\varepsilon,\mb s})(f)-\pi(f)}
\leq
\bb{1+2K}K_f(1+t)^D e^{-\delta t/2}.
\end{equation*}
This holds for every~$K\geq0$ and~$t\geq0$. Here
$p^-=\max\cb{2p-1,0}$, so the covered initial laws include every~$\mu_u$ with
$u\geq p^-$ and every point mass on a configuration with finitely many sites
in state~$0$.
If~$p\leq\frac12$, then~$\pi$ is the unique invariant measure and
\begin{equation*}
\sup_{\eta\in\Omega}
\abs{P_t^{\varepsilon,\mb s}f(\eta)-\pi(f)}
\leq
K_f(1+t)^D e^{-\delta t/2}.
\end{equation*}
Its monomial moments are given by~\eqref{eq:full-patch-limit-moments}.
\end{enumerate}
\end{proposition}

\begin{proof}
The two patch-positivity propositions and
Theorem~\ref{thm:patch-positivity-order} give part~\textup{(i)}. For
part~\textup{(ii)}, the added refresh supplies independent~$1\to0$ transitions
at rate at least~$\delta$, and hence a pure-death component of rate~$\delta$.
Apply Theorem~\ref{thm:common-invariant-limit} and
Corollary~\ref{cor:uniform-ergodicity}.
\end{proof}
