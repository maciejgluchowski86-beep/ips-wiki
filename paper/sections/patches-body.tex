% BLURB A. Why retain a successful-interaction skeleton
%
% Reader question:
%   What information from the full graphical construction should we condition
%   on so that all interactions between sites are fixed while local randomness
%   remains available to average?
%
% Job:
%   Motivate the skeleton as coarse conditioning information. Explain that an
%   actual interaction between distinct sites occurs exactly when a nonempty-
%   target mark rings at an active source. To determine such an interaction it
%   is enough to retain its source, time, and target. Deliberately forget its
%   kind: for fixed source/time/target, birth and split involve the same sites
%   and differ only in the resulting state of the source. Also omit deaths and
%   inactive-source rings. Once these records are fixed, their times divide each
%   site line into intervals within which the omitted randomness is local.
%
% Keep out:
%   The formal definition of I_T, the patch labels, and the finite-horizon
%   truncation. Do not claim a formal minimality theorem for the sigma-algebra.
% ============================================================================

We now choose a coarse~$\sigma$-algebra for the graphical construction on which
to condition. It should determine every interaction between distinct sites
while leaving local randomness unobserved. An interaction between sites occurs
exactly when a mark with nonempty target rings at an active source. To specify
such an interaction, it is enough to retain its source, time, and target. We do
not retain its kind~$\alpha$: for fixed~$(i,t,S)$, a birth and a split involve
the same source and target sites and differ only in the resulting state of the
source. We also omit deaths, whose target is empty, and rings at inactive
sources, which do not change the process at all. The retained records form the
successful-interaction skeleton. Once this skeleton is fixed, its interaction
times divide each site line into consecutive intervals, called patches. The
omitted deaths and inactive-source rings occur within individual site lines
between these interaction times, while the omitted kind of a retained
interaction affects only its source line.

% ============================================================================
% BLURB B. Successful interactions, skeleton, and conditioning information
%
% Reader question:
%   Exactly which marks are retained, what does the skeleton remember and
%   forget, and what coarse information is encoded by G_T?
%
% Job:
%   Define a successful interaction in I^P by
%       S != empty and i in A_{t-}.
%   Declare the deterministic initial interaction successful as well. Define its
%   record and the record (i,t,S) of a successful interaction. Emphasize that
%   the record deliberately forgets alpha: it remembers source, time, and
%   target, but not whether the mark was a split or a birth. Deaths and
%   inactive-source rings are also omitted.
%
%   Define I_T, I, and G_T = sigma(Y_0,I_T). Say that G_T records the initial
%   state and this coarse interaction information up to time T. Use the local
%   finiteness from the graphical construction to note that I_T is finite a.s.
%
% Formatting:
%   No paragraph heading is needed.
% ============================================================================

\begin{definition}[Successful interaction and its record]
\label{def:successful-interaction}
The deterministic initial interaction~$\iota_0$ from
\eqref{eq:initial-interaction} is declared successful and has record
\begin{equation*}
(\infty,0,A_0).
\end{equation*}
An interaction~$(i,t,\alpha,S)\in I^{\mathrm P}$ is
\emph{successful} when
\begin{equation*}
S\neq\varnothing
\qquad\text{and}\qquad
i\in A_{t-}.
\end{equation*}
Its successful-interaction record is~$(i,t,S)$. Thus the record remembers the
source, time, and target of the interaction, but not its kind~$\alpha$.
\end{definition}

For~$T<\infty$, let~$\mathcal I_T$ be the successful-interaction skeleton up to
time~$T$:
\begin{equation*}
\mathcal I_T
=
\cb{(\infty,0,A_0)}
\cup
\cb{(i,t,S):t\leq T\text{ and, for some }\alpha\in\cb{\beta,\delta},
(i,t,\alpha,S)\text{ is successful}}.
\end{equation*}
Thus~$\mathcal I_T$ contains exactly the retained records described above. Set
\begin{equation*}
\mathcal I=\bigcup_{T<\infty}\mathcal I_T,
\qquad
\mathcal G_T=\sigma\bb{Y_0,\mathcal I_T}.
\end{equation*}
Thus~$\mathcal G_T$ records the initial state and the successful-interaction
skeleton up to time~$T$. The local-finiteness statement in
Section~\ref{subsec:dual-graphical-construction} implies that~$\mathcal I_T$ is
finite almost surely.

% BLURB C. Patch geometry and boundary labels
%
% Reader question:
%   Given the successful-interaction skeleton, what is a patch geometrically,
%   and what do the I/O/E labels mean?
%
% Job:
%   Give the geometric picture first: on each site line, successful interactions
%   cut time into consecutive intervals. Each interval starts at an incoming or
%   outgoing successful interaction and ends at the next successful interaction
%   involving that site, if one exists.
%
%   Use one convention for all records: a record (j,t,S) is incoming at every
%   site in S and, when j is a site of Lambda, outgoing at j. This automatically
%   treats the initial record as supplying incoming time-zero boundaries at A_0.
%
%   Then define T^I(i), T^O(i), T(i), and e_i(s). Define a patch P and its data
%   i(P), [s(P),e(P)), S(P), and boundary labels X(P),Y(P). Explain in prose
%   before or immediately after the formulas:
%       I = incoming interaction,
%       O = outgoing interaction,
%       E = no later successful interaction on that site line.
%   Define the full family P and state that an E-terminal patch in the infinite
%   family is infinite, while I/O-terminal patches are finite.
%
% Formatting:
%   Keep the formal patch definition environment. A compact table may be kept
%   as a glossary if it still improves readability, but it should not merely
%   duplicate unexplained formulas. Do not use a \paragraph heading.
% ============================================================================

We regard a record~$(j,t,S)\in\mathcal I$ as incoming at every site in~$S$ and as outgoing at~$j$. On each site line in spacetime, these interactions
divide time into consecutive intervals. Each interval begins with an incoming
or outgoing interaction and ends at the next successful interaction involving
that site, if there is one.

For~$i\in\Lambda$, let
\begin{align*}
\mathcal T^{\mathsf I}(i)
&=
\cb{t\geq0:(j,t,S)\in\mathcal I
  \text{ for some }j,S\text{ with }i\in S},
\\
\mathcal T^{\mathsf O}(i)
&=
\cb{t>0:(i,t,S)\in\mathcal I\text{ for some }S},
\end{align*}
and set
\begin{equation*}
\mathcal T(i)=\mathcal T^{\mathsf I}(i)\cup\mathcal T^{\mathsf O}(i).
\end{equation*}
For~$s\in\mathcal T(i)$, the next successful interaction involving~$i$ occurs
at
\begin{equation*}
e_i(s)=\inf\cb{u>s:u\in\mathcal T(i)},
\qquad
\inf\varnothing=\infty.
\end{equation*}

\begin{definition}[Patch]
\label{def:patch}
Let~$(j,s,S)\in\mathcal I$ be a record for which~$i=j$ or~$i\in S$. The patch
beginning at~$(i,s)$ is
\begin{equation*}
P=
\bb{i(P),[s(P),e(P)),\mathsf X(P)\mathsf Y(P),S(P)},
\end{equation*}
where
\begin{equation*}
i(P)=i,
\qquad
s(P)=s,
\qquad
e(P)=e_i(s),
\qquad
S(P)=S.
\end{equation*}
The initial label~$\mathsf X(P)$ records whether the interaction at~$s(P)$ is
incoming or outgoing:
\begin{equation*}
\mathsf X(P)=
\pw{\mathsf I}{i(P)\in S(P)}{\mathsf O}{i(P)\notin S(P)}.
\end{equation*}
The terminal label~$\mathsf Y(P)$ records the next successful interaction on
the same site line, with~$\mathsf E$ meaning that there is no such interaction:
\begin{equation*}
\mathsf Y(P)=
\pwthree
{\mathsf I}{e(P)<\infty\text{ and }e(P)\in\mathcal T^{\mathsf I}\bb{i(P)}}
{\mathsf O}{e(P)<\infty\text{ and }e(P)\in\mathcal T^{\mathsf O}\bb{i(P)}}
{\mathsf E}{e(P)=\infty}.
\end{equation*}
\end{definition}

Let~$\mathcal P$ be the family of all patches. For this infinite-horizon family,
a patch with terminal label~$\mathsf E$ is infinite, while a patch with terminal
label~$\mathsf I$ or~$\mathsf O$ is finite.
