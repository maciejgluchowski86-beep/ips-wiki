\section{Main results}
\label{sec:main-results}

The first theorem is formulated using the signed monomial dual constructed in
Section~\ref{sec:signed-dual}. Starting from a finite set~$A$, the dual tracks
the sites on which the monomial~$\chi_A$ depends. Its graphical construction
records the interactions through which this dependence propagates between
sites. For each site, consecutive such interactions delimit time intervals;
these intervals, together with their boundary data, are the patches defined in
Section~\ref{sec:patches}. At time~$t$, let~$\mathcal B_t$ be the finite family
of bulk patches ending by time~$t$ and~$\mathcal E_t$ the finite family of
patches extending to time~$t$. A bulk patch~$P$ has contribution~$C(P)$. An end
patch based at~$i(P)$ has a contribution~$C(z,P)$ which is affine in
$z\in[0,1]$.

\begin{maintheorem}[Patch representation]
\label{thm:patch-representation}
For every~$A\Subset\Lambda$,~$t\geq0$, and~$\eta\in\Omega$,
\begin{equation}
\label{eq:patch-representation}
P_t\chi_A(\eta)
=
\E_A\Cb{
\prod_{P\in\mathcal B_t}C(P)
\prod_{P\in\mathcal E_t}C\bb{\eta\bb{i(P)},P}
}.
\end{equation}
For any initial law~$\mu$,
\begin{equation}
\label{eq:patch-representation-general-law}
(\mu P_t)(\chi_A)
=
\E_A\Cb{
\prod_{P\in\mathcal B_t}C(P)\,
\mu\bb{
\prod_{P\in\mathcal E_t}C\bb{\eta\bb{i(P)},P}
}
}.
\end{equation}
\end{maintheorem}

A spin system is patch-positive when~$C(P)\geq0$ for every possible bulk
patch~$P$. End-patch contributions also depend on the terminal spin. For a
patch-positive system, define the \emph{patch threshold profile}
$\mb p^\star=(p_i^\star)_{i\in\Lambda}$ by
\begin{equation*}
p_i^\star
=
\inf\cb{
p\in[0,1]:
C(p,P)\geq0
\text{ for every end patch based at }i
}.
\end{equation*}
Thus~$p_i^\star$ is the smallest terminal value for which every end-patch
contribution based at~$i$ is nonnegative. It is not an ergodicity or
phase-transition threshold. Define
\begin{equation*}
\chi_A^*(\eta)
=
\prod_{i\in A}\bb{\eta(i)-p_i^\star},
\qquad
\chi_\vn^*=1,
\end{equation*}
and, for probability measures~$\mu$ and~$\nu$, write~$\mu\preceq_*\nu$ when
\begin{equation*}
\mu\bb{\chi_A^*}\leq\nu\bb{\chi_A^*}
\qquad
\text{for every }A\Subset\Lambda.
\end{equation*}
On the full space of probability measures,~$\preceq_*$ need not have a largest
or smallest element; we therefore keep it distinct from ordinary stochastic
order.

We also use the class of probability measures whose centered moments are
nonnegative:
\begin{equation}
\label{eq:m-star-definition}
\mathcal M_*
=
\cb{
\mu:\mu\bb{\chi_A^*}\geq0
\text{ for every }A\Subset\Lambda
}.
\end{equation}
In particular,~$\mu_{\mb p}\in\mathcal M_*$ exactly when
$\mb p\geq\mb p^\star$.

\begin{maintheorem}[Patch positivity and centered-moment order]
\label{thm:patch-positivity-order}
\leavevmode
\begin{enumerate}[label=\textup{(\roman*)}]
\item Put~$r_i=c_i^0(\vn)+c_i^1(\vn)$. The spin system is patch-positive if and
only if, at every site~$i$ with~$r_i=0$, one has~$c_i\equiv0$, while at every
site with~$r_i>0$, the inequalities
\begin{equation}
\label{eq:principal-coefficient-criterion}
c_i^0(S)+c_i^1(S)\leq0,
\qquad
c_i^1(\vn)c_i^0(S)\geq c_i^0(\vn)c_i^1(S)
\end{equation}
hold for every nonempty~$S\subseteq N(i)$. When these conditions hold, the patch
threshold profile is
\begin{equation}
\label{eq:principal-threshold-profile}
p_i^\star
=
\max\cb{
0,
\sup_{\substack{
\vn\neq S\subseteq N(i)\\
c_i^0(S)+c_i^1(S)<0
}}
\frac{c_i^0(S)}{c_i^0(S)+c_i^1(S)}
}.
\end{equation}

\item The semigroup preserves the centered-moment order:
\begin{equation}
\label{eq:principal-centered-order}
\mu\preceq_*\nu
\quad\Longrightarrow\quad
\mu P_t\preceq_*\nu P_t
\qquad
\text{for every }t\geq0.
\end{equation}
No assumption that~$\mu$ or~$\nu$ belongs to~$\mathcal M_*$ is required in
\eqref{eq:principal-centered-order}. The semigroup also preserves
$\mathcal M_*$, and
\begin{equation*}
(\mu P_t)(\chi_A)\leq(\nu P_t)(\chi_A)
\end{equation*}
whenever~$\mu\preceq_*\nu$. In particular, for every~$\mu\in\mathcal M_*$,
\begin{equation*}
(\mu_{\mb p^\star}P_t)(\chi_A)
\leq
(\mu P_t)(\chi_A).
\end{equation*}
\end{enumerate}
\end{maintheorem}

\begin{corollary}[Pure-death comparison]
\label{cor:pure-death-comparison}
Let~$\Lcal$ and~$\Lcal'$ be spin-system generators. Suppose that~$\Lcal$ is
patch-positive and that
\begin{equation}
\label{eq:pure-death-decomposition}
\Lcal f(\eta)
=
\Lcal'f(\eta)
+
\sum_{i\in\Lambda}d_i
\bb{f\bb{\eta^{i,0}}-f(\eta)},
\qquad d_i\geq0.
\end{equation}
Thus~$\Lcal'$ is obtained by removing the environment-independent~$1\to0$
transitions with rates~$d_i$. The generator~$\Lcal'$ is patch-positive with
the same patch threshold profile~$\mb p^\star$, and its semigroup~$P_t'$
satisfies
\begin{equation}
\label{eq:principal-pure-death-comparison}
(\mu P_t)(\chi_A)
\leq
(\mu P_t')(\chi_A)
\qquad
\text{for every }\mu\in\mathcal M_*.
\end{equation}
\end{corollary}

The convergence estimate is first proved uniformly for initial laws in
$\mathcal M_*$. By linearity in the initial law, it extends to measures that
belong to~$\mathcal M_*$ after mixing with~$\mu_{\mb 1}$. For~$K\geq0$, set
\begin{equation}
\label{eq:m-minus-affine-characterization}
\mathcal M_{-,K}
=
\cb{
\mu:
\frac{\mu+K\mu_{\mb 1}}{1+K}\in\mathcal M_*
},
\qquad
\mathcal M_-
=
\bigcup_{K\geq0}\mathcal M_{-,K}.
\end{equation}
Thus~$\mathcal M_{-,0}=\mathcal M_*$. If
$p_i^-=\max\cb{2p_i^\star-1,0}$, then a Bernoulli product measure
$\mu_{\mb p}$ belongs to~$\mathcal M_{-,1}$ exactly when
$\mb p\geq\mb p^-$.

\begin{maintheorem}[Common invariant limit]
\label{thm:common-invariant-limit}
Suppose that~$\Lambda$ has polynomial growth of exponent~$D$, the spin system
is patch-positive, and it contains a pure-death component of
rate~$\varepsilon>0$. Then there is an invariant probability measure~$\pi$ such
that, for every local function~$f$, there is~$K_f<\infty$ satisfying
\begin{equation}
\label{eq:uniform-m-minus-convergence}
\sup_{\mu\in\mathcal M_{-,K}}
\abs{(\mu P_t)(f)-\pi(f)}
\leq
(1+2K)K_f(1+t)^D e^{-\varepsilon t/2}
\end{equation}
for every~$K\geq0$ and~$t\geq0$. Consequently,
$\mu P_t\Rightarrow\pi$ for every~$\mu\in\mathcal M_-$. Moreover,
\begin{equation}
\label{eq:full-patch-limit-moments}
\pi(\chi_A)
=
\E_A\Cb{
\prod_{P\in\mathcal P}C(P)
\ind\bb{\abs{\mathcal P}<\infty}
}
\qquad
\text{for every }A\Subset\Lambda.
\end{equation}
\end{maintheorem}

Under the hypotheses of Theorem~\ref{thm:common-invariant-limit}, every point
mass on a configuration with finitely many sites in state~$0$ belongs
to~$\mathcal M_-$. Hence the theorem gives convergence from every such
configuration.

\begin{corollary}[Uniform exponential ergodicity]
\label{cor:uniform-ergodicity}
Under the hypotheses of Theorem~\ref{thm:common-invariant-limit}, suppose also
that~$\mb p^\star\leq\mbs{\tfrac12}$. Then~$\pi$ is the unique invariant
measure and, for every local~$f$,
\begin{equation}
\label{eq:uniform-pointwise-convergence}
\sup_{\eta\in\Omega}
\abs{P_tf(\eta)-\pi(f)}
\leq
K_f(1+t)^D e^{-\varepsilon t/2}.
\end{equation}
\end{corollary}
