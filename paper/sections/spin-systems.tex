\section{Setup}
\label{sec:setup}
\label{sec:spin-systems}

\subsection{State space}
\label{subsec:lattices-observables}

Let~$\Lambda$ be a countable set. For each~$i\in\Lambda$, fix a finite
neighbourhood~$N(i)\subseteq\Lambda\setminus\cb{i}$, and assume
\begin{equation*}
\sup_{i\in\Lambda}\abs{N(i)}<\infty.
\end{equation*}
The neighbourhoods may be directed:~$j\in N(i)$ need not imply~$i\in N(j)$.
Write~$N_*(i)=N(i)\cup\cb{i}$ and, for~$A\subseteq\Lambda$,
\begin{equation*}
N_*(A)=\bigcup_{i\in A}N_*(i),
\qquad
B(A,0)=A,
\qquad
B(A,n+1)=N_*\bb{B(A,n)}.
\end{equation*}
Thus~$B(A,n)=N_*^n(A)$ is the directed neighbourhood ball reached from~$A$ in
at most~$n$ steps, and~$B(i,n)=B(\cb{i},n)$. We say that~$\Lambda$ has
\emph{polynomial growth of exponent~$D$} if there is~$K<\infty$ such that
\begin{equation}
\label{eq:polynomial-growth}
\abs{B(i,n)}\leq K(1+n)^D
\qquad
\text{for every }i\in\Lambda\text{ and }n\in\N.
\end{equation}
No symmetry of the neighbourhood relation is required.

We write~$A\Subset\Lambda$ when~$A$ is finite. The configuration space is
$\Omega=\cb{0,1}^{\Lambda}$. Following the standard KCSM convention
\cite{HartarskyToninelli2025}, state~$0$ is facilitating and state~$1$ is calm.

\begin{definition}[Local functions]
\label{def:local-function}
A function~$f:\Omega\to\R$ is \emph{local} if there is~$A\Subset\Lambda$ such
that~$f(\eta)=f(\xi)$ whenever~$\eta|_A=\xi|_A$. Such a set~$A$ is called a
dependence set of~$f$.
\end{definition}

\begin{definition}[Monomials]
\label{def:monomials}
For~$A\Subset\Lambda$, define
\begin{equation*}
\chi_A(\eta)=\prod_{i\in A}\eta(i),
\qquad
\chi_\varnothing=1.
\end{equation*}
If a local function depends on~$A$, it has a unique expansion in the
basis~$\cb{\chi_B:B\subseteq A}$.
\end{definition}

For a profile~$\mb a=(a_i)_{i\in\Lambda}\in[0,1]^\Lambda$, let~$\mu_{\mb a}$
be the Bernoulli product measure with one-density~$a_i$ at site~$i$. For a
constant~$a$, write~$\mb a=(a)_{i\in\Lambda}$ and~$\mu_a=\mu_{\mb a}$.

\subsection{Spin systems}
\label{subsec:rates-coefficients}
\label{subsec:calm-facilitating-rates}

For~$i\in\Lambda$ and~$a\in\cb{0,1}$, let~$\eta^{i,a}$ be obtained from~$\eta$
by setting the spin at~$i$ equal to~$a$. We write~$\eta^i$ for the
configuration obtained by flipping that spin.

\begin{definition}[Spin system]
\label{def:spin-system}
A spin system on~$\Omega$ is a continuous-time Markov process whose generator
acts on local functions by
\begin{equation}
\label{eq:spin-generator}
\Lcal f(\eta)
=
\sum_{i\in\Lambda}c_i(\eta)
\bb{f(\eta^i)-f(\eta)},
\end{equation}
where~$c_i(\eta)\geq0$ depends only on~$\eta|_{N(i)\cup\cb{i}}$ and
\begin{equation*}
\sup_{i\in\Lambda,\,\eta\in\Omega}c_i(\eta)<\infty.
\end{equation*}
\end{definition}

For~$x\in\cb{0,1}$, set~$c_i^x(\eta)=c_i(\eta^{i,x})$. Then~$c_i^x$ depends
only on~$\eta|_{N(i)}$, and
\begin{equation*}
c_i(\eta)
=
\bb{1-\eta(i)}c_i^0(\eta)+\eta(i)c_i^1(\eta).
\end{equation*}
Thus~$c_i^0$ is the rate of the flip~$0\to1$, while~$c_i^1$ is the rate of the
flip~$1\to0$. Each~$c_i^x$ has the unique multilinear expansion
\begin{equation}
\label{eq:rate-expansion}
c_i^x(\eta)
=
\sum_{S\subseteq N(i)}c_i^x(S)\chi_S(\eta).
\end{equation}
The same symbol denotes the function and its coefficients; the argument~$\eta$
or~$S$ distinguishes them. The coefficients are given by M\"obius inversion.

Under these assumptions, the standard Harris construction gives a Feller
process; we use this without further comment and refer to
\cite[Chapter~I]{Liggett1985}. Its semigroup is denoted by~$(P_t)_{t\geq0}$,
with~$P_tf(\eta)=\E_\eta\Cb{f(\eta_t)}$. For a probability measure~$\mu$, we
write~$(\mu P_t)(f)=\mu(P_tf)$.

\subsection{Independent noise}
\label{subsec:noise-ergodicity}
\label{def:bernoulli-refresh}
\label{def:bernoulli-noise}
\label{def:independent-noise}

For~$p\in[0,1]$, write~$q=1-p$. The Bernoulli refresh operator at site~$i$ is
\begin{align*}
E_{i,p}f(\eta)
&=
pf(\eta^{i,1})+qf(\eta^{i,0}),
\\
\bb{E_{i,p}-I}f(\eta)
&=
\bb{p\bb{1-\eta(i)}+q\eta(i)}
\bb{f(\eta^i)-f(\eta)}.
\end{align*}
Here~$p$ is the density of calm states and~$q$ the density of facilitating
states. In the KCSM and BABP literature, the prefactor multiplying
$E_{i,p}-I$ is often called the constraint or refresh rate and denoted by~$c_i$.
In this paper,~$c_i$ always denotes the resulting flip rate.

For a profile~$\mb s=(s_i)_{i\in\Lambda}\in[0,1]^\Lambda$, define the unit-rate
independent refresh generator
\begin{equation*}
\Ncal^{\mb s}f
=
\sum_{i\in\Lambda}\bb{E_{i,s_i}-I}f.
\end{equation*}
For~$\varepsilon\geq0$, the corresponding perturbation of~$\Lcal$ is
\begin{equation*}
\Lcal^{\varepsilon,\mb s}
=
\Lcal+\varepsilon\Ncal^{\mb s}.
\end{equation*}
Thus~$\mb s$ specifies the refresh law and~$\varepsilon$ is the amount of
independent noise added. The constant profile~$\mb 0$ gives pure deaths:
\begin{equation*}
\Ncal^{\mb 0}f(\eta)
=
\sum_{i\in\Lambda}\bb{f\bb{\eta^{i,0}}-f(\eta)}.
\end{equation*}
We say that~$\Lcal$ contains pure-death noise at rate~$\varepsilon>0$ if
\begin{equation*}
\Lcal=\Lcal^\varepsilon+\varepsilon\Ncal^{\mb 0}
\end{equation*}
for another spin-system generator~$\Lcal^\varepsilon$. Equivalently,
$c_i^1(\eta)\geq\varepsilon$ for every~$i$ and~$\eta$. The perturbation
$\Lcal^{\varepsilon,\mb s}$ contains pure-death noise at rate
\begin{equation*}
\delta=\varepsilon\inf_{i\in\Lambda}\bb{1-s_i}
\end{equation*}
whenever~$\delta>0$.

\subsection{Convergence}
\label{subsec:invariant-ergodicity}

\begin{definition}[Invariant measure]
\label{def:invariant-measure}
A probability measure~$\nu$ on~$\Omega$ is \emph{invariant} if
\begin{equation*}
\nu P_t=\nu
\qquad
\text{for every }t\geq0.
\end{equation*}
\end{definition}

\begin{definition}[Ergodicity]
\label{def:ergodicity}
The spin system is \emph{ergodic} if there is a probability measure~$\nu$ such
that
\begin{equation*}
\mu P_t\Rightarrow\nu
\qquad
\text{as }t\to\infty
\end{equation*}
for every probability measure~$\mu$ on~$\Omega$. The limiting measure is then
the unique invariant measure.
\end{definition}

Since~$\Omega$ is compact in the product topology, weak convergence is
determined by local functions.

\begin{definition}[Uniform exponential ergodicity]
\label{def:uniform-exponential-ergodicity}
The spin system is \emph{uniformly exponentially ergodic} if it has an
invariant measure~$\nu$, and there is~$\gamma>0$ such that for every local
function~$f$ there is~$C_f<\infty$ satisfying
\begin{equation*}
\sup_{\eta\in\Omega}
\abs{P_tf(\eta)-\nu(f)}
\leq
C_f e^{-\gamma t}
\qquad
\text{for every }t\geq0.
\end{equation*}
\end{definition}

Some results below are uniform only over specified classes of initial laws.
Such a restricted estimate is not called ergodicity unless it holds for every
point mass, and hence, by integration, for every initial law.
