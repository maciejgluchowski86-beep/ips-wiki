\section{Patch contributions and representation}
\label{sec:representation}

\begin{discussion}
  \item Explain that the Feynman--Kac sign and potential factor exactly
  over patches before any averaging.
  \item Explain why conditional averaging replaces the signed dual
  history by a product of one-site quantities.
  \item Preview that the affine dependence of end factors on the
  terminal spin connects the representation to the centered classes
  in \cref{def:centered-classes}.
\end{discussion}

\subsection{Patch contributions}

\begin{definition}[Patch contributions]
\label{def:patch-contribution}
Let \(P\) be a labeled patch and \(X^P\) its local active indicator. If
\(P\) begins at an outgoing interaction, let
\(\alpha_P\in\{\delta,\beta\}\) be its hidden kind and set
\[
  \sigma_P=\sigma_{i(P)}^{\alpha_P}(S(P));
\]
for an incoming patch, set \(\sigma_P=1\).

For a full patch, define
\begin{equation}
  C(P)
  =
  \E_P^{\mathrm{con}}\left[
    \sigma_P
    \exp\left(
      V_{i(P)}
      \int_{s(P)}^{e(P)}X_u^P\,\dd u
    \right)
  \right],
  \label{eq:full-contribution}
\end{equation}
using the endpoint limit when \(e(P)=\infty\). If \(P\) is an end
patch at time \(t\), define, for \(z\in[0,1]\),
\begin{equation}
  C(z,P)
  =
  \E_P^{\mathrm{con}}\left[
    \sigma_P
    \exp\left(
      V_{i(P)}
      \int_{s(P)}^tX_u^P\,\dd u
    \right)
    z^{X_t^P}
  \right].
  \label{eq:end-contribution}
\end{equation}
\end{definition}

\begin{proposition}[Affine end factors]
\label{prop:affine-end-factors}
Every end factor \(C(z,P)\) is affine in \(z\). The full formulas for
all four patch types are given in
\cref{prop:explicit-patch-factors}.
\end{proposition}

\begin{proof}
This is immediate from \eqref{eq:end-contribution}, since
\(X_t^P\in\{0,1\}\). The explicit coefficients follow from
\cref{prop:explicit-patch-factors}.
\end{proof}

\subsection{Random-product representation}

\begin{notation}
\label{not:skeleton-expectation}
Write \(\E_A\) for expectation over the successful-interaction
skeleton of the signed dual started from \((A,+)\).
\end{notation}

\begin{theorem}[Patch representation]
\label{thm:patch-representation}
For every \(A\Subset\Lambda\), \(t\ge0\), and
\(\xi\in\{0,1\}^{\Lambda}\),
\begin{equation}
  P_t\chi_A(\xi)
  =
  \E_A\left[
    \prod_{P\in\cB_t}C(P)
    \prod_{P\in\cE_t}C(\xi(i(P)),P)
  \right].
  \label{eq:patch-representation}
\end{equation}
\end{theorem}

\begin{proof}
The patches partition the active site-time set up to time \(t\).
Every successful nonempty-target interaction contributes its sign to
exactly one outgoing-initial patch, and every end patch records the
dual state at its site at time \(t\). Hence the integrand in
\eqref{eq:fk-duality} is
\begin{align*}
  &\sigma_t
  \exp\left(\int_0^tV(A_u)\,\dd u\right)
  \chi_{A_t}(\xi)
  \\
  &\quad=
  \prod_{P\in\cB_t}
  \left[
    \sigma_P
    \exp\left(
      V_{i(P)}
      \int_{s(P)}^{e(P)}X_u^P\,\dd u
    \right)
  \right]
  \prod_{P\in\cE_t}
  \left[
    \sigma_P
    \exp\left(
      V_{i(P)}
      \int_{s(P)}^tX_u^P\,\dd u
    \right)
    \xi(i(P))^{X_t^P}
  \right].
\end{align*}
Apply \cref{thm:patch-factorization} conditionally on \(\cG_t\), and
then take expectation.
\end{proof}

\begin{proposition}[General initial law and centered expansion]
\label{prop:general-law-representation}
For every probability measure \(\mu\),
\begin{equation}
  \mu(P_t\chi_A)
  =
  \E_A\left[
    \prod_{P\in\cB_t}C(P)\,
    \mu\left(
      \prod_{P\in\cE_t}C(\eta(i(P)),P)
    \right)
  \right].
  \label{eq:general-law-representation}
\end{equation}
Suppose a profile \(\pstar\) satisfies
\begin{equation}
  C(z,P)
  =
  C(p_{i(P)}^\star,P)
  +
  b(P)\bigl(z-p_{i(P)}^\star\bigr),
  \qquad
  C(p_{i(P)}^\star,P)\ge0,\quad b(P)\ge0.
  \label{eq:centered-end-factor}
\end{equation}
Then
\begin{align}
  &\mu\left(
    \prod_{P\in\cE_t}C(\eta(i(P)),P)
  \right)
  \notag\\
  &\quad=
  \sum_{\cQ\subseteq\cE_t}
  \mu\left(
    \chi_{\{i(P):P\in\cQ\}}^\star
  \right)
  \prod_{P\in\cQ}b(P)
  \prod_{P\in\cE_t\setminus\cQ}
  C(p_{i(P)}^\star,P).
  \label{eq:end-factor-expansion}
\end{align}
\end{proposition}

\begin{proof}
Integrating \eqref{eq:patch-representation} gives
\eqref{eq:general-law-representation}. Distinct end patches are based
at distinct sites by \cref{def:finite-patch-families}. Expanding the
product in \eqref{eq:centered-end-factor} therefore gives
\eqref{eq:end-factor-expansion}.
\end{proof}
