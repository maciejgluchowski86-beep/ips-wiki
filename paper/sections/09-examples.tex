\subsection{FA-1f}

\begin{definition}[FA-1f]
\label{def:fa-1f}
Let \(q\in(0,1)\) be the vacancy density and \(p=1-q\). For a site
\(i\) with finite nonempty neighbour set \(N(i)\), set
\[
  c_i(\eta)
  =
  1-\chi_{N(i)}(\eta).
\]
At rate \(c_i(\eta)\), refresh \(\eta(i)\) from the Bernoulli law with
zero-density \(q\).
\end{definition}

\begin{proposition}[Patch parameters for FA-1f]
\label{prop:fa-patch-parameters}
FA-1f has \patchpositivity\ and
\[
  p_i^\star=p.
\]
Its only nonzero coefficients are
\[
  c_i^0(\vn)=p,
  \qquad
  c_i^1(\vn)=q,
  \qquad
  c_i^0(N(i))=-p,
  \qquad
  c_i^1(N(i))=-q.
\]
\end{proposition}

\begin{proof}
The nonempty coefficient sum is \(-1\), while the determinant in
\eqref{eq:patch-positive-nondegenerate} is zero. Apply
\cref{thm:patch-positivity-criterion,prop:critical-density-formula}.
\end{proof}

\begin{definition}[Soft FA-1f]
\label{def:soft-fa-1f}
For \(s\in(0,1)\), the soft FA-1f generator is
\begin{equation}
  \cL_{\mathrm{sFA}}f(\eta)
  =
  \sum_{i\in\Lambda}
  \bigl[s+(1-s)(1-\chi_{N(i)}(\eta))\bigr]
  \bigl(E_i^qf(\eta)-f(\eta)\bigr),
  \label{eq:soft-fa-generator}
\end{equation}
where \(E_i^q\) refreshes the spin at \(i\) from the Bernoulli law
with zero-density \(q\).
\end{definition}

\begin{corollary}[Soft FA-1f]
\label{cor:soft-fa}
Let \(\Lambda\) have polynomial growth exponent \(D\). For every local
\(f\),
\[
  \sup_{\nu\in\Mminus}
  \abs{\nu(P_tf)-\mu_q(f)}
  \le
  K_f(1+t)^D e^{-qst/2}.
\]
Here \(\pstar=p\mathbf1\), so a product initial law with one-density
profile \(\mathbf p_0\) belongs to \(\Mminus\) precisely when
\[
  p_{0,i}\ge(2p-1)\vee0
  \qquad(i\in\Lambda).
\]
If \(p\le1/2\), the system has
\uniformexponentialergodicity.
\end{corollary}

\begin{proof}
The nonempty coefficients in \cref{prop:fa-patch-parameters} are
multiplied by \(1-s\), so \patchpositivity\ and \(p^\star=p\) remain
unchanged. The \(1\)-to-\(0\) rate is at least \(qs\), and
\(\mu_q\) is an \invariantmeasure. Apply
\cref{thm:common-limit,cor:uniform-ergodicity}.
\end{proof}

\subsection{The biased annihilating branching process}

\begin{definition}[Biased annihilating branching process]
\label{def:babp}
On a lattice with finite neighbour sets, BABP refreshes site \(i\)
from the Bernoulli law with zero-density \(q\) at rate
\[
  c_i(\eta)
  =
  \sum_{j\in N(i)}(1-\eta(j)).
\]
\end{definition}

\begin{proposition}[Patch parameters for BABP]
\label{prop:babp-patch-parameters}
BABP has \patchpositivity\ and
\[
  p_i^\star=p=1-q.
\]
Its only nonzero coefficients are
\begin{align*}
  c_i^0(\vn)&=p\abs{N(i)},
  &
  c_i^1(\vn)&=q\abs{N(i)},
  \\
  c_i^0(\{j\})&=-p,
  &
  c_i^1(\{j\})&=-q,
  \qquad j\in N(i).
\end{align*}
\end{proposition}

\begin{proof}
Each nonempty target is a singleton. For every \(j\in N(i)\), the
coefficient sum is \(-1\), and the determinant in
\eqref{eq:patch-positive-nondegenerate} is zero. Apply
\cref{thm:patch-positivity-criterion,prop:critical-density-formula}.
\end{proof}

\begin{citedresult}[Homogeneous BABP convergence
  {\cite[Application 2 and Section 5.1.1]{martinelli-shapira-toninelli2025}}]
\label{cited:babp-homogeneous}
Let BABP run on \(\Z^d\), and let \(\mu_{q_0}\) be the homogeneous
Bernoulli product law with vacancy density \(q_0>0\). For every local
function \(f\), there are \(c>0\) and \(C_f<\infty\) such that
\[
  \abs{\mu_{q_0}(P_tf)-\mu_q(f)}
  \le C_fe^{-ct}.
\]
\end{citedresult}

\begin{citedresult}[Inhomogeneous quasi-thinning range
  {\cite[Remark 5.5]{martinelli-shapira-toninelli2025}}]
\label{cited:babp-quasi-thinning}
If the initial one-density profile satisfies
\[
  p_{0,i}\le\sqrt p
  \qquad(i\in\Z^d),
\]
then \(\mu_{\mathbf p_0}P_t\) converges exponentially to \(\mu_q\) on
local functions.
\end{citedresult}

\begin{proposition}[High-density BABP convergence]
\label{prop:babp-high-density}
Suppose that
\[
  p\le p_{0,i}\le p_+<1
  \qquad(i\in\Z^d).
\]
Then \(\mu_{\mathbf p_0}P_t\) converges exponentially to \(\mu_q\) on
local functions. More generally, the same conclusion holds for every
\(\nu\in\Mstar\) such that
\begin{equation}
  0
  \le
  \nu(\chi_B^\star)
  \le
  (p_+-p)^{\abs B}
  \qquad(B\Subset\Z^d).
  \label{eq:babp-centered-upper-bound}
\end{equation}
\end{proposition}

\begin{proof}
Let \(q_-=1-p_+>0\). For every nonempty finite \(B\),
\[
  \mu_q(\chi_B^\star)=0,
  \qquad
  \mu_{\mathbf p_0}(\chi_B^\star)
  =
  \prod_{i\in B}(p_{0,i}-p),
  \qquad
  \mu_{q_-}(\chi_B^\star)
  =
  (p_+-p)^{\abs B}.
\]
\Cref{thm:centered-order} gives
\[
  p^{\abs A}
  =
  \mu_q(P_t\chi_A)
  \le
  \mu_{\mathbf p_0}(P_t\chi_A)
  \le
  \mu_{q_-}(P_t\chi_A).
\]
The last term converges exponentially to \(p^{\abs A}\) by
\cref{cited:babp-homogeneous}. The same squeeze applies under
\eqref{eq:babp-centered-upper-bound}. Expand a local function in the
finite monomial basis.
\end{proof}

  \item Compare \cref{prop:babp-high-density} precisely with

  is not covered by quasi-thinning.
  \item Explain why \cref{thm:common-limit} does not apply to
  unperturbed BABP: the all-one configuration is absorbing, so there
  is no uniform pure-death component.
  \item Do not claim full \ergodicity: both \(\mu_q\) and the point mass
  at the all-one configuration are \invariantmeasures.
\end{discussion}
