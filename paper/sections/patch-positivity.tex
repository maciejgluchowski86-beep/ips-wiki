\section{Proof of Theorem~\ref{thm:patch-positivity-order}}
\label{sec:patch-positivity}

This section proves Theorem~\ref{thm:patch-positivity-order}. The first two
subsections prove part~\textup{(i)}: we first characterize patch positivity in
terms of the M\"obius coefficients of~$\lambda_i$ and~$\lambda_i\rho_i$, and
then compute the patch threshold profile. After interpreting the criterion, we
return to the patch representation to prove part~\textup{(ii)}, preservation of
the centered-moment order. The remaining subsections give consequences of the
theorem. The starting point is the representation from
Theorem~\ref{thm:patch-representation}:
\begin{equation*}
(\mu P_t)(\chi_A)
=
\E_A\Cb{
\prod_{P\in\mathcal B_t}C(P)\,
\mu\bb{
\prod_{P\in\mathcal E_t}C\bb{\eta\bb{i(P)},P}
}
}.
\end{equation*}
The patch contributions appearing here are defined in
Section~\ref{sec:representation}. Patch positivity means that, for every
$A\Subset\Lambda$, almost surely every patch~$P$ in the full patch
family~$\mathcal P$ satisfies~$C(P)\geq0$. The end contributions require an
additional sign control because~$C(z,P)$ still depends on its terminal
argument~$z$.

\subsection{Coefficient criterion}
\label{subsec:positivity-criterion}

% ============================================================================
% BLURB A. Reduce patch positivity to the distinct contribution formulas
%
% Reader question:
%   Patch positivity asks for C(P)>=0 for every patch in the full patch family.
%   How can this be checked from the local rates?
%
% Job:
%   Begin the proof of part (i) of Theorem~\ref{thm:patch-positivity-order}.
%   Fix a site i and a patch P in the full patch family based at i, and write
%   Delta=e(P)-s(P), allowing Delta=infinity. Recall the initial and terminal
%   boundary labels. Explain that terminal labels I and E give the same
%   contribution formula, so the six boundary labels reduce to four distinct
%   formulas: {II,IE}, IO, {OI,OE}, and OO.
%
%   Show first that every patch whose initial boundary is incoming has a
%   nonnegative contribution. State the consequence clearly: only patches whose
%   initial boundary is outgoing can impose conditions on the coefficients of
%   the flip rates.
%
% Keep out:
%   Derivation of the patch-contribution formulas, interpretation of the
%   coefficient inequalities, and finite-time end-patch positivity.
% ============================================================================

We begin the proof of part~\textup{(i)} of
Theorem~\ref{thm:patch-positivity-order}. Fix a
site~$i$ and a patch~$P\in\mathcal P$ based at~$i$, and write
$\Delta=e(P)-s(P)\in(0,\infty]$. The initial label
$\mathsf X(P)\in\cb{\mathsf I,\mathsf O}$ records whether the interaction at
$s(P)$ is incoming or outgoing at~$i$. The terminal label
$\mathsf Y(P)\in\cb{\mathsf I,\mathsf O,\mathsf E}$ records whether the next
successful interaction is incoming or outgoing, with~$\mathsf E$ meaning that
there is no later successful interaction at that site. The contributions~$C(P)$
for~$\mathsf{XI}$ and~$\mathsf{XE}$ patches are the same, so there are four
distinct cases:~$\cb{\mathsf{II},\mathsf{IE}}$,~$\mathsf{IO}$,
$\cb{\mathsf{OI},\mathsf{OE}}$, and~$\mathsf{OO}$.

If the initial boundary is incoming, Appendix~\ref{app:bulk-formulas} gives
\begin{equation*}
C(P)
=
\begin{cases}
\displaystyle
\frac{\psi_i(\Delta,1)}{\varphi_i(\Delta)},
&\mathsf X(P)\mathsf Y(P)\in\cb{\mathsf{II},\mathsf{IE}},
\\[1em]
e^{V_i\Delta},
&\mathsf X(P)\mathsf Y(P)=\mathsf{IO}.
\end{cases}
\end{equation*}
At~$\Delta=\infty$, the first expression is read by taking the corresponding
limit. The coefficient of~$c_i$ corresponding to death at~$i$ is
$c_i^0(\vn)=c_i(\mb 0)\geq0$, so deaths have positive sign. Hence
$\psi_i(\Delta,1)\geq0$, while~$\varphi_i(\Delta)>0$ for every realized patch,
and the exponential is positive. Thus every patch with an incoming initial
boundary has a nonnegative contribution. Only patches with an outgoing initial
boundary can therefore impose conditions on the coefficients of the flip
rates.

% ============================================================================
% BLURB B. Outgoing terminal boundary gives the first inequality
%
% Reader question:
%   What restriction comes from a patch whose initial and terminal boundaries
%   are both outgoing?
%
% Job:
%   Let the initial outgoing interaction have nonempty target S. For an OO
%   patch, quote the formula
%       C(P)=sigma_i^beta(S)e^{V_i Delta}
%   together with
%       beta_i(S)sigma_i^beta(S)=-c_i^0(S)-c_i^1(S).
%   Conclude that all such contributions are nonnegative exactly when
%       c_i^0(S)+c_i^1(S)<=0.
%
% Reader takeaway:
%   The OO formula gives exactly the first coefficient inequality.
% ============================================================================

Suppose now that the initial boundary is outgoing, with nonempty target~$S$.
If the terminal boundary is also outgoing, then
Appendix~\ref{app:bulk-formulas} gives
\begin{equation*}
C(P)=\sigma_i^\beta(S)e^{V_i\Delta},
\qquad
\beta_i(S)\sigma_i^\beta(S)
=-c_i^0(S)-c_i^1(S).
\end{equation*}
Thus the sign of an~$\mathsf{OO}$ contribution is the sign of
$-c_i^0(S)-c_i^1(S)$. Consequently, all such contributions are nonnegative
exactly when
\begin{equation}
\label{eq:patch-positivity-first-inequality}
c_i^0(S)+c_i^1(S)\leq0
\qquad\text{for every nonempty }S\subseteq N(i).
\end{equation}

% ============================================================================
% BLURB C. The other outgoing-initial formula gives the second inequality
%
% Reader question:
%   Once the first inequality holds, what further restriction comes from the
%   common OI/OE contribution formula?
%
% Job:
%   State that OI and OE have the same contribution formula. Its denominator is
%   positive, so the sign is determined by
%       c_i^0(S)-(c_i^0(S)+c_i^1(S))psi_i(Delta,1).
%   Put r_i=c_i^0(emptyset)+c_i^1(emptyset), and treat here the case r_i>0.
%   Use the explicit formula for psi_i(Delta,1). Under the first inequality,
%   explain that the numerator is minimized at Delta=infinity. Compute it and
%   obtain the second coefficient inequality.
% ============================================================================

The remaining outgoing-initial cases,~$\mathsf{OI}$ and~$\mathsf{OE}$, have
the same contribution formula. Its denominator is positive, while its numerator
is
\begin{equation*}
c_i^0(S)-\bb{c_i^0(S)+c_i^1(S)}\psi_i(\Delta,1).
\end{equation*}
Put~$r_i=c_i^0(\vn)+c_i^1(\vn)$. Note that
$r_i=c_i(\mb 0)+c_i(\mb 0^i)\geq0$, and suppose first that~$r_i>0$. Then
\begin{equation*}
\psi_i(\Delta,1)
=
\frac{c_i^0(\vn)}{r_i}
+
\frac{c_i^1(\vn)}{r_i}e^{-r_i\Delta}.
\end{equation*}
Under~\eqref{eq:patch-positivity-first-inequality}, the numerator is
nonincreasing in~$\Delta$. Its minimum is therefore the value at
$\Delta=\infty$, which is
\begin{equation*}
\frac{
c_i^1(\vn)c_i^0(S)-c_i^0(\vn)c_i^1(S)
}{r_i}.
\end{equation*}
Hence the~$\mathsf{OI}/\mathsf{OE}$ contribution is nonnegative for
every patch length exactly when
\begin{equation}
\label{eq:patch-positivity-second-inequality}
c_i^1(\vn)c_i^0(S)
\geq
c_i^0(\vn)c_i^1(S)
\qquad\text{for every nonempty }S\subseteq N(i).
\end{equation}

% ============================================================================
% BLURB D. The case r_i=0 and completion of the coefficient criterion
%
% Reader question:
%   What happens when r_i=0, and have the four distinct contribution formulas
%   proved both directions of the coefficient criterion in part (i)?
%
% Job:
%   When r_i=0, use psi_i(Delta,1)=1. The OI/OE formula then gives
%   c_i^1(S)<=0 for every nonempty S. Use the multilinear expansion and the
%   nonnegativity of the rate function to deduce c_i^1\equiv0. The OO inequality
%   then similarly forces c_i^0\equiv0.
%
%   Close the converse explicitly using the four distinct contribution formulas.
% ============================================================================

It remains to consider~$r_i=0$. Since both terms in~$r_i$ are nonnegative,
$c_i^0(\vn)=c_i^1(\vn)=0$, and
Appendix~\ref{app:positivity-algebra} gives~$\psi_i(\Delta,1)=1$. The numerator
of the~$\mathsf{OI}/\mathsf{OE}$ contribution therefore reduces to
$-c_i^1(S)$, so patch positivity requires~$c_i^1(S)\leq0$ for every nonempty
$S\subseteq N(i)$. For a neighbourhood configuration whose set of~$1$'s is
$T\subseteq N(i)$, the expansion~\eqref{eq:rate-expansion} then gives
\begin{equation*}
c_i^1(\eta)
=
\sum_{S\subseteq T}c_i^1(S)
\leq0.
\end{equation*}
Since~$c_i^1$ is a nonnegative rate function,~$c_i^1\equiv0$. The inequality
from the~$\mathsf{OO}$ contribution now gives~$c_i^0(S)\leq0$ for every
nonempty~$S$, and the same argument, using~$c_i^0(\vn)=0$, yields
$c_i^0\equiv0$.

Conversely, suppose first that~$r_i>0$ and that
\eqref{eq:patch-positivity-first-inequality} and
\eqref{eq:patch-positivity-second-inequality} hold for every nonempty
$S\subseteq N(i)$. The incoming-initial formulas are nonnegative, the
$\mathsf{OO}$ formula is nonnegative by the first inequality, and the common
$\mathsf{OI}/\mathsf{OE}$ formula is nonnegative by the second. These are all
four distinct contribution formulas. If~$r_i=0$ and~$c_i\equiv0$, there are no
outgoing interactions from~$i$, while the incoming-initial contributions are
nonnegative by the first part of the proof. Since~$i$ was arbitrary, this
proves the coefficient criterion in part~\textup{(i)} of
Theorem~\ref{thm:patch-positivity-order}.

\subsection{Patch threshold}
\label{subsec:critical-density}

% ============================================================================
% BLURB A. Reduce end-patch positivity to the OE case
%
% Reader question:
%   The coefficient criterion has characterized patch positivity. What remains
%   to prove in part (i), and which end patches can determine the threshold?
%
% Job:
%   State that it remains to compute the threshold profile in part (i).
%   Recall that p_i^star is the smallest p for which every end patch based at i
%   has C(p,P)>=0. Every end patch has terminal label E, so it has type IE or OE.
%   Use the end-contribution formula to show that IE patches are nonnegative for
%   every p in [0,1]. Hence only OE patches can determine p_i^star.
%
% Reader takeaway:
%   The remaining part of Theorem B(i) is the OE threshold calculation.
% ============================================================================

To complete part~\textup{(i)} of Theorem~\ref{thm:patch-positivity-order}, it
remains to compute the patch threshold profile. Patch positivity controls
$C(P)$ for the full patch family, while the finite-time representation also
contains end contributions~$C(p,P)$. Recall that~$p_i^\star$ is the smallest
$p\in[0,1]$ such that~$C(p,P)\geq0$ for every end patch based at~$i$. Every end
patch has terminal label~$\mathsf E$, so it has type~$\mathsf{IE}$ or
$\mathsf{OE}$.

Fix an end patch~$P$ based at~$i$ and write~$\Delta=e(P)-s(P)$. For a
$\mathsf{IE}$ patch,~\eqref{eq:end-contribution-formulas} gives
\begin{equation*}
C(p,P)=\frac{\psi_i(\Delta,p)}{\varphi_i(\Delta)}.
\end{equation*}
For~$p\in[0,1]$,~$\psi_i(\Delta,p)\geq0$ and~$\varphi_i(\Delta)>0$. Thus
$\mathsf{IE}$ patches impose no restriction on~$p$, and~$p_i^\star$ is
determined entirely by~$\mathsf{OE}$ patches.

% ============================================================================
% BLURB B. Positivity of one outgoing end patch
%
% Reader question:
%   For a fixed OE end patch with initial target S, how large must p be to make
%   its contribution nonnegative for every possible patch length?
%
% Job:
%   Recall r_i=c_i^0(emptyset)+c_i^1(emptyset). Dispose of r_i=0 in one sentence,
%   then assume r_i>0. Since the denominator is positive, define the numerator
%       N_{i,S}(Delta,p)
%       =c_i^0(S)-(c_i^0(S)+c_i^1(S))psi_i(Delta,p).
%   Keep this notation: it is used throughout the calculation.
%
%   Patch positivity gives c_i^0(S)+c_i^1(S)<=0, so N is nondecreasing in p.
%   Write the explicit formula for psi_i(Delta,p), and compute the two endpoint
%   values in Delta. The Delta->infinity endpoint is the determinant expression
%   from 5.1 divided by r_i and is therefore nonnegative.
%
%   Since N is affine in exp(-r_i Delta), positivity for every Delta>=0 is
%   equivalent to N_{i,S}(0,p)>=0.
%
% Reader takeaway:
%   The second coefficient inequality already controls the large-Delta endpoint;
%   the threshold comes entirely from the Delta=0 endpoint.
% ============================================================================

Recall that~$r_i=c_i^0(\vn)+c_i^1(\vn)$. If~$r_i=0$, the coefficient criterion
gives~$c_i\equiv0$ and hence~$p_i^\star=0$. Suppose therefore that~$r_i>0$,
and let~$P$ be an~$\mathsf{OE}$ end patch based at~$i$, whose initial outgoing
interaction has nonempty target~$S$. By~\eqref{eq:end-contribution-formulas},
the denominator of~$C(p,P)$ is positive, while its numerator is
\begin{equation*}
N_{i,S}(\Delta,p)
=
c_i^0(S)-\bb{c_i^0(S)+c_i^1(S)}\psi_i(\Delta,p).
\end{equation*}
Patch positivity gives~$c_i^0(S)+c_i^1(S)\leq0$, so
$N_{i,S}(\Delta,p)$ is nondecreasing in~$p$. Moreover,
\begin{equation*}
\psi_i(\Delta,p)
=
\frac{c_i^0(\vn)}{r_i}
+
\bb{p-\frac{c_i^0(\vn)}{r_i}}e^{-r_i\Delta}.
\end{equation*}
Consequently,~$N_{i,S}(\Delta,p)$ is affine in~$e^{-r_i\Delta}$, with endpoint
values
\begin{align*}
N_{i,S}(0,p)
&=
c_i^0(S)-\bb{c_i^0(S)+c_i^1(S)}p,
\\
\lim_{\Delta\to\infty}N_{i,S}(\Delta,p)
&=
\frac{
c_i^1(\vn)c_i^0(S)-c_i^0(\vn)c_i^1(S)
}{r_i}.
\end{align*}
The second endpoint is nonnegative by
\eqref{eq:patch-positivity-second-inequality}. It follows that the
$\mathsf{OE}$ contribution is nonnegative for every patch length exactly when
\begin{equation*}
N_{i,S}(0,p)\geq0.
\end{equation*}

% ============================================================================
% BLURB C. Threshold formula and empty-neighbour bound
%
% Reader question:
%   What threshold results after imposing the Delta=0 condition for every target
%   S, and how large can that threshold be?
%
% Job:
%   Split according to c_i^0(S)+c_i^1(S). If the sum vanishes, show that S
%   imposes no restriction. If it is negative, solve N_{i,S}(0,p)>=0 and use the
%   second coefficient inequality to show that the resulting ratio is at most
%   c_i^0(emptyset)/r_i<=1.
%
%   Take the supremum over the targets with negative sum and impose p>=0. State
%   explicitly that this completes part (i) of the theorem. Then record the
%   threshold bound as a useful consequence.
%
% Keep out:
%   Dynamic interpretation of the coefficient inequalities; that belongs in
%   Subsection~\ref{subsec:positivity-interpretation}. Centered end-factor
%   expansions and moment comparisons belong in the following subsections.
% ============================================================================

If~$c_i^0(S)+c_i^1(S)=0$, then
\eqref{eq:patch-positivity-second-inequality}, together with
$c_i^1(S)=-c_i^0(S)$, gives~$r_i c_i^0(S)\geq0$. Hence~$c_i^0(S)\geq0$ and
$N_{i,S}(0,p)\geq0$ for every~$p\in[0,1]$. Such a target therefore imposes no
restriction on the threshold. If instead~$c_i^0(S)+c_i^1(S)<0$, then
$N_{i,S}(0,p)\geq0$ is equivalent to
\begin{equation*}
p\geq
\frac{c_i^0(S)}{c_i^0(S)+c_i^1(S)}.
\end{equation*}
For every such~$S$, the second coefficient inequality can also be rewritten as
\begin{equation*}
r_i c_i^0(S)
\geq
c_i^0(\vn)\bb{c_i^0(S)+c_i^1(S)}.
\end{equation*}
Since the factor in parentheses is negative,
\begin{equation*}
\frac{c_i^0(S)}{c_i^0(S)+c_i^1(S)}
\leq
\frac{c_i^0(\vn)}{r_i}
\leq1.
\end{equation*}
If there is no nonempty~$S\subseteq N(i)$ with
$c_i^0(S)+c_i^1(S)<0$, then~$p_i^\star=0$. Otherwise, taking the supremum over
all such targets and imposing~$p\geq0$ gives
\begin{equation*}
p_i^\star
=
\max\cb{
0,
\sup_{\substack{
\vn\neq S\subseteq N(i)\\
c_i^0(S)+c_i^1(S)<0
}}
\frac{c_i^0(S)}{c_i^0(S)+c_i^1(S)}
}.
\end{equation*}
This proves the threshold formula and completes the proof of
part~\textup{(i)} of Theorem~\ref{thm:patch-positivity-order}.

\begin{corollary}[Threshold bound]
\label{cor:critical-profile-empty-bound}
If~$r_i = 0$ then~$p_i^* = 0$. If~$r_i>0$, then
\begin{equation*}
p_i^\star
\leq
\frac{c_i(\mb 0)}{c_i(\mb 0)+c_i(\mb 0^i)}\leq 1.
\end{equation*}
\end{corollary}

\begin{proof}
If~$r_i=0$ then the patch-positivity criterion gives~$c_i\equiv 0$ and hence~$p_i^*=0$. For the case~$r_i>0$, for every target~$S$ appearing in the supremum defining~$p_i^\star$, the preceding
calculation gives
\begin{equation*}
\frac{c_i^0(S)}{c_i^0(S)+c_i^1(S)}
\leq
\frac{c_i^0(\vn)}{r_i} = \frac{c_i(\mb 0)}{c_i(\mb 0)+c_i(\mb 0^i)}.
\end{equation*}
Taking the supremum and then the maximum with~$0$ proves the claim.
\end{proof}
