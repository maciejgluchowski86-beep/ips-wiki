\section{Proof of Theorem~\ref{thm:common-invariant-limit}}
\label{sec:convergence}

The main estimate compares the system with the one obtained by removing pure
deaths at rate~$\varepsilon$. Patch positivity makes both patch weights
nonnegative, and the removed deaths increase the weights. A
successful interaction after a cut time~$T$ has a backward chain of
outgoing-terminal patches whose total lifetime exceeds~$T$, producing a factor
$e^{-\varepsilon T}$. If there is no successful interaction after~$T$, the
probability-weighted end factors present at~$T$ relax to limits independent of
the initial measure~$\mu$ at rate~$e^{-\varepsilon(t-T)}$. Restricting the
pre-$T$ skeleton to a directed ball bounds the number of end patches. Undoing
duality bounds the weight outside this ball by a zero-boundary finite-volume
error, which is controlled by finite propagation.

\subsection{Comparison weights}
\label{subsec:positive-skeleton-weights}

Remove deaths at rate~$\varepsilon$ from every site and write
\begin{equation}
\label{eq:uniform-death-decomposition}
\Lcal
=
\Lcal^\varepsilon
+
\varepsilon\Ncal^{\mb 0}.
\end{equation}
Let~$P_t^\varepsilon$ and~$C^\varepsilon$ denote the semigroup and patch
contributions of~$\Lcal^\varepsilon$. The two systems have the same signed dual,
successful-interaction skeleton, and reference patch laws. By
Theorem~\ref{thm:patch-positivity-order},~$\Lcal^\varepsilon$ has patch
positivity and the same threshold profile~$\mb p^\star$.

Fix~$A\Subset\Lambda$. For~$\mu\in\mathcal M_*$, define
\begin{equation}
\label{eq:finite-horizon-weight}
W_t^\mu
=
\prod_{P\in\mathcal B_t}C(P)\,
\mu\bb{
 \prod_{P\in\mathcal E_t}
 C\bb{\eta\bb{i(P)},P}
},
\end{equation}
and define~$W_t^{\varepsilon,\mu}$ by replacing each contribution by its
$\Lcal^\varepsilon$ counterpart. Define also
\begin{align*}
W
&=
\prod_{P\in\mathcal P}C(P)
\ind\bb{\abs{\mathcal P}<\infty},
\\
W^\varepsilon
&=
\prod_{P\in\mathcal P}C^\varepsilon(P)
\ind\bb{\abs{\mathcal P}<\infty}.
\end{align*}
The weights for~$\Lcal^\varepsilon$ give a nonnegative integrable upper bound
for the weights of~$\Lcal$. On a patch which remains dual-active until an
outgoing terminal interaction, the ratio of the two contributions is the
survival factor over the lifetime of the patch. Multiplying these factors along
a backward chain gives the exponential estimate used below.

\begin{lemma}[Nonnegative weights]
\label{lem:positive-weights}
For every~$\mu\in\mathcal M_*$ and~$t\geq0$,
\begin{equation}
\label{eq:positive-weight-comparison}
0\leq W_t^\mu\leq W_t^{\varepsilon,\mu},
\qquad
\E_A\Cb{W_t^{\varepsilon,\mu}}
=
(\mu P_t^\varepsilon)(\chi_A)
\leq1.
\end{equation}
Moreover,
\begin{equation}
\label{eq:limiting-weight-mass}
0\leq W\leq W^\varepsilon,
\qquad
\E_A\Cb{W^\varepsilon}\leq1.
\end{equation}
If a patch~$P$ has outgoing terminal boundary, then
\begin{equation}
\label{eq:outgoing-terminal-exact-factor}
C(P)
=
e^{-\varepsilon\bb{e(P)-s(P)}}C^\varepsilon(P).
\end{equation}
\end{lemma}

Let
\begin{equation*}
\mb{Cone}_T
=
\bigcup_{(i,u,S)\in\mathcal I_T}
\bb{\cb{i}\cup S}\setminus\cb{\infty}
\end{equation*}
be the set of sites reached by the skeleton by time~$T$. For
$A\subseteq R\Subset\Lambda$, set
\begin{equation*}
E_T^R=\cb{\mb{Cone}_T\subseteq R},
\qquad
\rho_A(T,R)
=
\norm{\bb{P_T-P_T^{R,0}}\chi_A}_\infty,
\end{equation*}
where~$P_t^{R,0}$ is the zero-boundary semigroup on~$R$. The total weight on
which the pre-$T$ skeleton leaves~$R$ is bounded by the error made when the
spin system is replaced by its zero-boundary restriction to~$R$. Finite
propagation makes this error exponentially small when~$R$ grows linearly with
$T$.

\begin{lemma}[Spatial confinement]
\label{lem:spatial-confinement}
For every~$\mu\in\mathcal M_*$,
\begin{align}
0
&\leq
\E_A\Cb{W_t^\mu\ind\bb{(E_T^R)^c}}
\leq
\rho_A(T,R),
\label{eq:finite-horizon-confinement-bound}
\\
0
&\leq
\E_A\Cb{W\ind\bb{(E_T^R)^c}}
\leq
\rho_A(T,R).
\label{eq:limiting-confinement-bound}
\end{align}
\end{lemma}

For~$0\leq T<t$, let~$L_{T,t}$ be the event that no successful interaction
occurs in~$(T,t]$, and let~$L_T$ be the event that none occurs after~$T$.
Recall
\begin{equation*}
r_i=c_i^0(\vn)+c_i^1(\vn),
\qquad
p_i^\circ=\frac{c_i^0(\vn)}{r_i};
\end{equation*}
on an interval with no nonempty-target interaction, the one-site process has
rates~$c_i^0(\vn)$ from~$0$ to~$1$ and~$c_i^1(\vn)$ from~$1$ to~$0$.
Thus~$p_i^\circ$ is its equilibrium calm-state density. The pure-death component
ensures~$r_i\geq\varepsilon$.

To control the event~$L_{T,t}^c$, begin with its first successful interaction
after~$T$ and trace its dependence backward. The resulting patches have
outgoing terminal boundaries and form a chain from time zero to the
interaction time. Their lifetimes therefore add rather than overlap.

\begin{lemma}[Backward chain of outgoing patches]
\label{lem:backward-outgoing-chain}
Let~$(i,u,S)$ be a successful-interaction record with~$u>0$. There are distinct
patches~$P_1,\ldots,P_n$, each with outgoing terminal boundary, such that
\begin{equation*}
s(P_1)=0,
\qquad
e(P_n)=u,
\qquad
e(P_k)=s(P_{k+1}),
\qquad
i(P_{k+1})\in N_*\bb{i(P_k)}
\end{equation*}
for~$1\leq k<n$. Consequently,
\begin{equation}
\label{eq:outgoing-chain-total-length}
\sum_{k=1}^n\bb{e(P_k)-s(P_k)}=u.
\end{equation}
\end{lemma}

The exact factor in Lemma~\ref{lem:positive-weights} multiplies along this
chain to~$e^{-\varepsilon u}$. Since~$u>T$, the weight of realizations with a
late successful interaction is at most~$e^{-\varepsilon T}$.

\begin{lemma}[Late interactions]
\label{lem:late-interactions}
For every~$\mu\in\mathcal M_*$,
\begin{align}
0
&\leq
\E_A\Cb{W_t^\mu\ind\bb{E_T^R\cap L_{T,t}^c}}
\leq
e^{-\varepsilon T},
\label{eq:finite-late-interaction-bound}
\\
0
&\leq
\E_A\Cb{W\ind\bb{E_T^R\cap L_T^c}}
\leq
e^{-\varepsilon T}.
\label{eq:limiting-late-interaction-bound}
\end{align}
\end{lemma}

It remains to compare the finite and infinite weights on the complementary
event, where the skeleton stops by time~$T$. The bulk factors are then fixed.
Only the end patches continue, and their dependence on the terminal
configuration relaxes through deaths. Each centered
slope decays at rate at least~$\varepsilon$, while confinement bounds the number
of end patches by~$\abs R$.

\begin{lemma}[No-late-interaction relaxation]
\label{lem:no-late-relaxation}
For~$A\subseteq R\Subset\Lambda$,~$\mu\in\mathcal M_*$, and~$0\leq T<t$,
\begin{equation}
\label{eq:no-late-relaxation-bound}
\abs{
 \E_A\Cb{W_t^\mu\ind\bb{E_T^R\cap L_{T,t}}}
 -
 \E_A\Cb{W\ind\bb{E_T^R\cap L_T}}
}
\leq
\bb{1+\abs R}e^{-\varepsilon\bb{t-T}}.
\end{equation}
\end{lemma}
\subsection{Completion of the proof}

\begin{proof}[Proof of Theorem~\ref{thm:common-invariant-limit}]
For~$\mu\in\mathcal M_*$, the patch representation and the definition of~$W$
give
\begin{align*}
(\mu P_t)(\chi_A)-\E_A\Cb{W}
={}&
\E_A\Cb{W_t^\mu\ind\bb{(E_T^R)^c}}
-
\E_A\Cb{W\ind\bb{(E_T^R)^c}}
\\
&+
\E_A\Cb{W_t^\mu\ind\bb{E_T^R\cap L_{T,t}^c}}
-
\E_A\Cb{W\ind\bb{E_T^R\cap L_T^c}}
\\
&+
\E_A\Cb{W_t^\mu\ind\bb{E_T^R\cap L_{T,t}}}
-
\E_A\Cb{W\ind\bb{E_T^R\cap L_T}}.
\end{align*}
The three estimates give
\begin{equation}
\label{eq:three-term-convergence-bound}
\abs{(\mu P_t)(\chi_A)-\E_A\Cb{W}}
\leq
2\rho_A(T,R)
+
2e^{-\varepsilon T}
+
\bb{1+\abs R}e^{-\varepsilon\bb{t-T}}.
\end{equation}
By Lemma~\ref{lem:finite-propagation}, there is~$v<\infty$ such that, for
$R_T=B\bb{A,\lceil vT\rceil}$,
\begin{equation*}
\rho_A(T,R_T)\leq C_Ae^{-\varepsilon T},
\qquad
\abs{R_T}\leq C_A(1+T)^D.
\end{equation*}
Taking~$T=t/2$ in~\eqref{eq:three-term-convergence-bound} gives
\begin{equation}
\label{eq:m-star-monomial-convergence}
\sup_{\mu\in\mathcal M_*}
\abs{(\mu P_t)(\chi_A)-\E_A\Cb{W}}
\leq
K_A(1+t)^D e^{-\varepsilon t/2}.
\end{equation}
Every local function is a finite linear combination of monomials, so the same
estimate holds for local~$f$, with the corresponding linear combination of the
limits in~\eqref{eq:full-patch-limit-moments}. In particular,
$\mu_{\mb 1}P_t$ has at most one subsequential weak limit. Compactness of
$\Omega$ gives a subsequential limit~$\pi$, and
\eqref{eq:m-star-monomial-convergence} identifies its monomial moments with
\eqref{eq:full-patch-limit-moments}. Hence~$\mu_{\mb 1}P_t\Rightarrow\pi$.
For local~$g$ and~$s\geq0$, the Feller property gives
\begin{equation*}
(\pi P_s)(g)
=
\lim_{t\to\infty}(\mu_{\mb 1}P_tP_s)(g)
=
\lim_{t\to\infty}(\mu_{\mb 1}P_{t+s})(g)
=
\pi(g),
\end{equation*}
so~$\pi$ is invariant.

Let~$\mu\in\mathcal M_{-,K}$ and set
\begin{equation*}
\overline\mu
=
\frac{\mu+K\mu_{\mb 1}}{1+K}
\in\mathcal M_*.
\end{equation*}
Since~$\mu=(1+K)\overline\mu-K\mu_{\mb 1}$,
\begin{align*}
\abs{(\mu P_t)(f)-\pi(f)}
&\leq
(1+K)\abs{(\overline\mu P_t)(f)-\pi(f)}
+
K\abs{(\mu_{\mb 1}P_t)(f)-\pi(f)}
\\
&\leq
(1+2K)K_f(1+t)^D e^{-\varepsilon t/2}.
\end{align*}
This proves~\eqref{eq:uniform-m-minus-convergence}.
\end{proof}

\begin{proof}[Proof of Corollary~\ref{cor:uniform-ergodicity}]
If~$\mb p^\star\leq\mbs{\tfrac12}$, then~$\mb p^-=\mb 0$ and every
probability measure belongs to~$\mathcal M_{-,1}$. Theorem
\ref{thm:common-invariant-limit} gives the result.
\end{proof}

\subsection{Proofs of the auxiliary estimates}
\label{subsec:convergence-estimate-proofs}

\begin{proof}[Proof of Lemma~\ref{lem:positive-weights}]
For~$\mathcal Q\subseteq\mathcal E_t$, write
$I(\mathcal Q)=\cb{i(P):P\in\mathcal Q}$. The centered end expansion gives
\begin{equation}
\label{eq:finite-weight-centered-expansion}
W_t^\mu
=
\prod_{P\in\mathcal B_t}C(P)
\sum_{\mathcal Q\subseteq\mathcal E_t}
\mu\bb{\chi_{I(\mathcal Q)}^*}
\prod_{P\in\mathcal Q}\kappa(P)
\prod_{P\in\mathcal E_t\setminus\mathcal Q}
C\bb{p_{i(P)}^\star,P}.
\end{equation}
For~$\mu\in\mathcal M_*$, every centered moment in this expression is
nonnegative. The patchwise pure-death comparison gives
\begin{align*}
0&\leq C(P)\leq C^\varepsilon(P),
&&P\in\mathcal B_t,
\\
0&\leq C\bb{p_{i(P)}^\star,P}
\leq C^\varepsilon\bb{p_{i(P)}^\star,P},
&&P\in\mathcal E_t,
\\
0&\leq\kappa(P)\leq\kappa^\varepsilon(P),
&&P\in\mathcal E_t.
\end{align*}
Thus every summand is nonnegative and bounded by the corresponding
$\varepsilon$ summand, proving the first inequality in
\eqref{eq:positive-weight-comparison}. The patch representation for
$\Lcal^\varepsilon$ gives
\begin{equation*}
\E_A\Cb{W_t^{\varepsilon,\mu}}
=
(\mu P_t^\varepsilon)(\chi_A)
\leq1.
\end{equation*}
Pointwise comparison gives~$0\leq W\leq W^\varepsilon$. On
$\cb{\abs{\mathcal P}<\infty}$, the all-one finite-horizon weights converge to
$W^\varepsilon$, while~$W^\varepsilon=0$ on the complementary event. Hence
\begin{equation*}
W^\varepsilon
\leq
\liminf_{u\to\infty}W_u^{\varepsilon,\mu_{\mb 1}},
\end{equation*}
and Fatou's lemma proves~\eqref{eq:limiting-weight-mass}.

If~$P$ has outgoing terminal boundary, consistency forces its site to remain
dual-active throughout~$[s(P),e(P))$. Removing the pure deaths changes the
one-site potential from~$V_i$ to~$V_i+\varepsilon$ and changes neither the sign
nor the consistency normalizer. Thus
\begin{equation*}
F(P)
=
e^{-\varepsilon\bb{e(P)-s(P)}}F^\varepsilon(P),
\end{equation*}
and taking the common consistent-patch expectation proves
\eqref{eq:outgoing-terminal-exact-factor}.
\end{proof}

\begin{proof}[Proof of Lemma~\ref{lem:spatial-confinement}]
Proposition~\ref{prop:confined-interaction-identity} gives
\begin{equation*}
\E_A\Cb{W_t^\mu\ind\bb{E_T^R}}
=
(\mu P_{t-T}P_T^{R,0})(\chi_A).
\end{equation*}
Subtracting this identity from
$\E_A[W_t^\mu]=(\mu P_{t-T}P_T)(\chi_A)$ yields
\begin{equation*}
\E_A\Cb{W_t^\mu\ind\bb{(E_T^R)^c}}
=
(\mu P_{t-T})\bb{\bb{P_T-P_T^{R,0}}\chi_A}.
\end{equation*}
The left-hand side is nonnegative, and the absolute value of the right-hand
side is at most~$\rho_A(T,R)$. This proves
\eqref{eq:finite-horizon-confinement-bound}. Apply the same bound with
$\mu=\mu_{\mb 1}$ and horizon~$u\geq T$. Since
$W_u^{\mu_{\mb 1}}\to W$ when~$\abs{\mathcal P}<\infty$ and~$W=0$ otherwise,
Fatou's lemma gives~\eqref{eq:limiting-confinement-bound}.
\end{proof}

\begin{proof}[Proof of Lemma~\ref{lem:backward-outgoing-chain}]
Start with the patch at the source~$i$ ending at~$u$. Its terminal boundary is
outgoing because~$i$ is dual-active immediately before the successful
interaction. Trace predecessors backwards. If the current patch begins with an
incoming interaction, take the patch at the source of that interaction ending
at the same time. If it begins with an outgoing interaction, consistency forces
that interaction to be a birth, so take the preceding patch at the same site.
In either case the predecessor has outgoing terminal boundary. The boundary
times decrease strictly, so local finiteness forces the construction to reach
time zero after finitely many steps. Reverse the resulting sequence. Consecutive
intervals then meet at their endpoints, and the later site is either the same
site or a target of the earlier interaction; hence
$i(P_{k+1})\in N_*\bb{i(P_k)}$. The intervals partition~$[0,u)$, which
proves~\eqref{eq:outgoing-chain-total-length}.
\end{proof}

\begin{proof}[Proof of Lemma~\ref{lem:late-interactions}]
On~$L_{T,t}^c$, let~$u\in(T,t]$ be the first successful-interaction time after
$T$, and let~$P_1,\ldots,P_n$ be its backward chain. Every patch in the chain
belongs to~$\mathcal B_t$, and
\begin{equation*}
\prod_{k=1}^n C(P_k)
=
e^{-\varepsilon u}
\prod_{k=1}^n C^\varepsilon(P_k)
\leq
e^{-\varepsilon T}
\prod_{k=1}^n C^\varepsilon(P_k).
\end{equation*}
Compare the centered expansions of~$W_t^\mu$ and
$W_t^{\varepsilon,\mu}$ term by term. Every remaining bulk contribution, end
constant, and end slope is bounded by its~$\varepsilon$ counterpart, and the
centered moment multiplying the term is nonnegative. Therefore
\begin{equation*}
W_t^\mu\ind\bb{L_{T,t}^c}
\leq
e^{-\varepsilon T}W_t^{\varepsilon,\mu}.
\end{equation*}
Taking expectations proves~\eqref{eq:finite-late-interaction-bound}. The same
chain comparison on~$\cb{\abs{\mathcal P}<\infty}\cap L_T^c$ gives
\begin{equation*}
W\ind\bb{L_T^c}
\leq
e^{-\varepsilon T}W^\varepsilon,
\end{equation*}
and~\eqref{eq:limiting-weight-mass} proves
\eqref{eq:limiting-late-interaction-bound}.
\end{proof}

\begin{proof}[Proof of Lemma~\ref{lem:no-late-relaxation}]
Proposition~\ref{prop:no-late-continuation-identity} gives
\begin{align*}
\CE{W_t^\mu\ind\bb{L_{T,t}}}{\mathcal G_T}
&=
\prod_{P\in\mathcal B_T}C(P)\,
\CP{L_{T,t}}{\mathcal G_T}
\mu\bb{
 \prod_{P\in\mathcal E_T}
 C\bb{\eta\bb{i(P)},P^{\uparrow t}}
},
\\
\CE{W\ind\bb{L_T}}{\mathcal G_T}
&=
\prod_{P\in\mathcal B_T}C(P)
\prod_{P\in\mathcal E_T}C\bb{p_{i(P)}^\circ,P}.
\end{align*}
Write~$B_T=\prod_{P\in\mathcal B_T}C(P)$. By
Lemma~\ref{lem:end-factor-relaxation}, the conditional absolute difference is
at most
\begin{align*}
&e^{-\varepsilon\bb{t-T}}B_T
\CP{L_{T,t}}{\mathcal G_T}
\mu\bb{
 \prod_{P\in\mathcal E_T}
 C^\varepsilon\bb{\eta\bb{i(P)},P^{\uparrow t}}
}
\\
&\qquad+
e^{-\varepsilon\bb{t-T}}B_T\abs{\mathcal E_T}
\prod_{P\in\mathcal E_T}C\bb{p_{i(P)}^\circ,P}.
\end{align*}
On~$E_T^R$, the map~$P\mapsto i(P)$ sends~$\mathcal E_T$ bijectively onto
$\mb{Cone}_T$, so~$\abs{\mathcal E_T}\leq\abs R$. Since
$B_T\leq B_T^\varepsilon:=\prod_{P\in\mathcal B_T}C^\varepsilon(P)$, the
conditional continuation identity for~$\Lcal^\varepsilon$ gives
\begin{align*}
&\E_A\Cb{
 \ind\bb{E_T^R}B_T
 \CP{L_{T,t}}{\mathcal G_T}
 \mu\bb{
  \prod_{P\in\mathcal E_T}
  C^\varepsilon\bb{\eta\bb{i(P)},P^{\uparrow t}}
 }
}
\\
&\qquad\leq
\E_A\Cb{
 \ind\bb{E_T^R}B_T^\varepsilon
 \CP{L_{T,t}}{\mathcal G_T}
 \mu\bb{
  \prod_{P\in\mathcal E_T}
  C^\varepsilon\bb{\eta\bb{i(P)},P^{\uparrow t}}
 }
}
\\
&\qquad=
\E_A\Cb{W_t^{\varepsilon,\mu}\ind\bb{E_T^R\cap L_{T,t}}}
\leq1.
\end{align*}
Likewise,
\begin{align*}
&\E_A\Cb{
 \ind\bb{E_T^R}B_T\abs{\mathcal E_T}
 \prod_{P\in\mathcal E_T}C\bb{p_{i(P)}^\circ,P}
}
\\
&\qquad\leq
\abs R\,
\E_A\Cb{
 \ind\bb{E_T^R}B_T
 \prod_{P\in\mathcal E_T}C\bb{p_{i(P)}^\circ,P}
}
\\
&\qquad=
\abs R\,
\E_A\Cb{W\ind\bb{E_T^R\cap L_T}}
\leq\abs R.
\end{align*}
Taking expectations and combining the two bounds proves
\eqref{eq:no-late-relaxation-bound}.
\end{proof}

The class~$\mathcal M_-$ enters only through the signs in the centered
expansion of the end factors. Outside this class the proof discards
cancellations between terms of different signs. It gives no obstruction to
convergence from other initial laws.
