\section{Comparison under removal of pure deaths}
\label{sec:pure-death-comparison}

For a nonnegative profile \(\mathbf a=(a_i)_{i\in\Lambda}\), let
\[
  \cD_{\mathbf a}f(\eta)
  =
  \sum_{i\in\Lambda}
  a_i\bigl(f(\eta^{i,0})-f(\eta)\bigr).
\]
This generator sets a spin to zero at an
environment-independent rate.

\begin{theorem}[Pure-death comparison]
\label{thm:pure-death-comparison}
Let \(\cL\) and \(\cL'\) be uniformly bounded finite-range generators
such that
\[
  \cL=\cL'+\cD_{\mathbf a}.
\]
Suppose that \(\cL\) has patch positivity.  Then \(\cL'\) also has
patch positivity and the same critical profile \(\pstar\).  If \(P_t\)
and \(P_t'\) are the corresponding semigroups, then
\begin{equation}
  \nu(P_t\chi_A)
  \le
  \nu(P_t'\chi_A)
  \qquad
  (\nu\in\Mstar,\ A\Subset\Lambda,\ t\ge0).
  \label{eq:pure-death-comparison}
\end{equation}
\end{theorem}

\begin{proof}
Removing the pure deaths changes only the empty-neighbour coefficient:
\[
  c_i^{\prime\,1}(\vn)=c_i^1(\vn)-a_i.
\]
All nonempty-neighbour coefficients, and hence the successful-
interaction skeleton, remain unchanged.  For nonempty
\(S\subseteq N(i)\), the determinant in the patch-positivity
criterion changes from
\[
  D_{i,S}
  =
  c_i^1(\vn)c_i^0(S)-c_i^0(\vn)c_i^1(S)
\]
to
\[
  D_{i,S}'=D_{i,S}-a_ic_i^0(S).
\]
If \(c_i^0(S)\le0\), this can only increase the determinant.  If
\(c_i^0(S)>0\), the first patch-positivity inequality gives
\(c_i^1(S)<0\), and the reduced determinant is nonnegative directly.
The degenerate case follows by the same two-sign check.  Thus
\(\cL'\) remains patch-positive.  Formula
\eqref{eq:critical-density-formula} uses only nonempty-neighbour
coefficients, so the critical profile is unchanged.

Let \(C,C'\) denote the two families of patch factors.  The explicit
formulas in \cref{app:patch-calculations} give
\[
  0\le C(P)\le C'(P)
\]
for every bulk patch.  For an end patch based at \(i\), expand around
the common critical value:
\[
  C(z,P)=C(p_i^\star,P)+b(P)(z-p_i^\star),
\]
and likewise for \(C'\).  The same formulas give
\begin{equation}
  0\le C(p_i^\star,P)\le C'(p_i^\star,P),
  \qquad
  0\le b(P)\le b'(P).
  \label{eq:centered-end-comparison}
\end{equation}
Every term in the centered expansion
\eqref{eq:end-factor-expansion} is nonnegative for
\(\nu\in\Mstar\), and \eqref{eq:centered-end-comparison} orders it by
the corresponding primed term.  The patch representation then proves
\eqref{eq:pure-death-comparison} for each skeleton before averaging.
\end{proof}

\begin{corollary}[Invariant-measure comparison]
\label{cor:invariant-comparison}
Under the assumptions of \cref{thm:pure-death-comparison}, suppose
that the two spin systems have unique invariant measures \(\pi\) and
\(\pi'\).  Then
\[
  \pi(\chi_A)\le\pi'(\chi_A)
  \qquad(A\Subset\Lambda).
\]
\end{corollary}

\begin{proof}
Apply \eqref{eq:pure-death-comparison} to
\(\mu_{\mathbf1}\in\Mstar\), average over \(t\in[0,T]\), and let
\(T\to\infty\).  Every subsequential Cesàro limit is invariant, so
uniqueness identifies the limits with \(\pi\) and \(\pi'\).
\end{proof}

\begin{remark}
The conclusion compares all joint occupation probabilities.  It does
not by itself imply stochastic domination.
\end{remark}
