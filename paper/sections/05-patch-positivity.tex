\section{Patch positivity and the critical profile}
\label{sec:patch-positivity}

\begin{discussion}
  \item Separate the two positivity questions: bulk factors are
  controlled by a local coefficient criterion, while end factors
  impose a threshold on the terminal one-density.
  \item Explain why the resulting profile \pstar\ is the natural
  centering profile for the measure classes in
  \cref{def:centered-classes}.
\end{discussion}

\subsection{Bulk positivity}

\begin{definition}[Patch positivity]
\label{def:patch-positivity}
The spin system has the patch positivity property if every bulk
contribution \(C(P)\) is nonnegative.
\end{definition}

\begin{theorem}[Local patch-positivity criterion]
\label{thm:patch-positivity-criterion}
All bulk contributions based at \(i\) are nonnegative if and only if,
for every nonempty \(S\subseteq N(i)\),
\begin{equation}
  \begin{cases}
  c_i^0(S)+c_i^1(S)\le0,\\
  c_i^1(\vn)c_i^0(S)-c_i^0(\vn)c_i^1(S)\ge0,
  \end{cases}
  \qquad
  \text{if }c_i^0(\vn)+c_i^1(\vn)>0,
  \label{eq:patch-positive-nondegenerate}
\end{equation}
whereas, if \(c_i^0(\vn)+c_i^1(\vn)=0\), the criterion is
\begin{equation}
  c_i^0(S)+c_i^1(S)\le0,
  \qquad
  c_i^1(S)\le0.
  \label{eq:patch-positive-degenerate}
\end{equation}
\end{theorem}

\begin{proof}
Check the four bulk rows in
\cref{prop:explicit-patch-factors}. The \(\sI\sI\) and
\(\sI\sO\) rows are automatically nonnegative. The \(\sO\sO\) row is
nonnegative exactly when \(c_i^0(S)+c_i^1(S)\le0\). Under this
condition, the numerator in the \(\sO\sI\) row is smallest at the
limiting empty-neighbour density. If the empty-neighbour refresh rate
is positive, its nonnegativity is exactly the determinant inequality
in \eqref{eq:patch-positive-nondegenerate}. If it vanishes, the
numerator reduces to \(-c_i^1(S)\).
\end{proof}

\subsection{Critical density}

\begin{definition}[Patch critical density]
\label{def:critical-profile}
For a patch-positive system, define
\[
  p_i^\star
  =
  \inf\set{
    p\in[0,1]:
    C(p,P)\ge0
    \text{ for every end patch based at }i
  }.
\]
The patch critical-density profile is
\(\pstar=(p_i^\star)_{i\in\Lambda}\).
\end{definition}

\begin{proposition}[Coefficient formula]
\label{prop:critical-density-formula}
For every \(i\in\Lambda\),
\begin{equation}
  p_i^\star
  =
  \max\left\{
    0,
    \sup_{\substack{\vn\ne S\subseteq N(i)\\
                    c_i^0(S)+c_i^1(S)<0}}
    \frac{c_i^0(S)}{c_i^0(S)+c_i^1(S)}
  \right\}.
  \label{eq:critical-density-formula}
\end{equation}
If
\[
  r_i=c_i^0(\vn)+c_i^1(\vn)>0,
  \qquad
  q_i=\frac{c_i^0(\vn)}{r_i},
\]
then \(p_i^\star\le q_i\).
\end{proposition}

\begin{proof}
The \(\sI\sE\) end factor is nonnegative for every terminal value. For
an \(\sO\sE\) patch with initial target \(S\), its numerator is
\[
  c_i^0(S)
  -
  \bigl(c_i^0(S)+c_i^1(S)\bigr)\psi_i(\Delta,p).
\]
Patch positivity makes this expression nondecreasing in \(p\). As the
patch length varies, its potentially smallest value is
\[
  c_i^0(S)
  -
  \bigl(c_i^0(S)+c_i^1(S)\bigr)p.
\]
Solving these inequalities over nonempty \(S\) gives
\eqref{eq:critical-density-formula}. The determinant inequality in
\eqref{eq:patch-positive-nondegenerate} gives \(p_i^\star\le q_i\).
\end{proof}

\subsection{Centered measure classes}

\begin{definition}[Centered monomials and measure classes]
\label{def:centered-classes}
For \(A\Subset\Lambda\), define
\begin{equation}
  \chi_A^\star(\eta)
  =
  \prod_{i\in A}\bigl(\eta(i)-p_i^\star\bigr),
  \qquad
  \chi_\vn^\star=1.
  \label{eq:centered-monomial}
\end{equation}
Define
\begin{align}
  \mathcal M_\star
  &=
  \set{
    \mu:
    \mu(\chi_A^\star)\ge0
    \text{ for every }A\Subset\Lambda
  },
  \label{eq:Mstar}
  \\
  \mathcal M_-
  &=
  \set{
    \mu:
    \mu(\chi_A^\star)
    \ge-\chi_A^\star(\mathbf1)
    \text{ for every }A\Subset\Lambda
  }.
  \label{eq:Mminus}
\end{align}
\end{definition}

\begin{proposition}[Elementary properties of the centered classes]
\label{prop:centered-classes}
The classes \(\Mstar\) and \(\Mminus\) are convex and weakly closed,
\(\Mstar\subseteq\Mminus\), and
\begin{equation}
  \mu\in\Mminus
  \quad\Longleftrightarrow\quad
  \frac{\mu+\mu_{\mathbf1}}2\in\Mstar.
  \label{eq:Mminus-affine}
\end{equation}
For a Bernoulli product law with one-density profile \(\mathbf p\),
\begin{align}
  \mu_{\mathbf p}\in\Mstar
  &\quad\Longleftrightarrow\quad
  \mathbf p\ge\pstar,
  \label{eq:product-Mstar}
  \\
  \mu_{\mathbf p}\in\Mminus
  &\quad\Longleftrightarrow\quad
  p_i\ge(2p_i^\star-1)\vee0
  \quad(i\in\Lambda).
  \label{eq:product-Mminus}
\end{align}
\end{proposition}

\begin{proof}
Convexity and weak closedness follow from the defining moment
inequalities. Since
\(\mu_{\mathbf1}(\chi_A^\star)=\chi_A^\star(\mathbf1)\),
\eqref{eq:Mminus-affine} follows by linearity. For a product law,
\[
  \mu_{\mathbf p}(\chi_A^\star)
  =
  \prod_{i\in A}(p_i-p_i^\star).
\]
The singleton inequalities give the necessity in
\eqref{eq:product-Mstar} and \eqref{eq:product-Mminus}; after normalization by
\(\chi_A^\star(\mathbf1)\), sufficiency for \(\Mminus\) follows because
a finite product of numbers in \([-1,1]\) is at least \(-1\).
\end{proof}

\begin{lemma}
\label{lem:all-measures-Mminus}
If \(\pstar\le\frac12\mathbf1\), then every probability measure on
\(\{0,1\}^{\Lambda}\) belongs to \(\Mminus\).
\end{lemma}

\begin{proof}
For every configuration \(\eta\),
\[
  \frac{\chi_A^\star(\eta)}
       {\chi_A^\star(\mathbf1)}
  =
  \prod_{i\in A}
  \frac{\eta(i)-p_i^\star}{1-p_i^\star}
  \ge-1.
\]
Integration gives \eqref{eq:Mminus}.
\end{proof}

\begin{remark}
\label{rem:centered-classes-not-mixtures}
Neither centered class is characterized by the corresponding family of
product measures or its mixtures.
\end{remark}
