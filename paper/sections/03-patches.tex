\section{Successful interactions and patches}
\label{sec:patches}

\begin{discussion}
  \item Explain which information is retained by the
  successful-interaction skeleton and which clock marks remain hidden.
  \item Use \cref{fig:patch-schematic} to explain why the retained
  interactions cut active site-lines into one-site intervals.
  \item Emphasize that the factorization theorem is finite-horizon and
  uses only ordinary Poisson disintegration on disjoint source-time
  intervals.
\end{discussion}

\subsection{The successful-interaction skeleton}

\begin{definition}[Successful interaction]
\label{def:successful-interaction}
A nonempty-target interaction \((i,t,\alpha,S)\), with
\(\alpha\in\{\delta,\beta\}\), is successful if
\[
  i\in A_{t-}.
\]
Its skeleton is \((i,t,S)\), which does not reveal whether
\(\alpha=\delta\) or \(\alpha=\beta\). The initial interaction in
\cref{con:dual-graphical} is declared successful and has skeleton
\((\infty,0,A)\).
\end{definition}

\begin{definition}[Finite-horizon skeleton]
\label{def:finite-skeleton}
For \(T<\infty\), let \(\cI_T\) consist of the initial skeleton and all
ordinary successful-interaction skeletons through time \(T\), and set
\begin{equation}
  \cG_T=\sigma\bigl((A_0,\sigma_0),\cI_T\bigr).
  \label{eq:skeleton-sigma-field}
\end{equation}
Empty-target deaths do not belong to \(\cI_T\).
\end{definition}

\subsection{Patches}

\begin{definition}[Patch]
\label{def:patch}
A successful interaction \((i,t,S)\) touches \(i\) as an outgoing
touch and each \(j\in S\) as an incoming touch. At each site, order
all successful touches in time. A patch is the labeled site-time
interval between two successive touches:
\[
  P
  =
  \{i(P)\}\times[s(P),e(P))
  \times\{\mathsf X(P)\mathsf Y(P)\}
  \times\{S(P)\}.
\]
Here \(S(P)\) is the target of the interaction at \(s(P)\);
\(\mathsf X(P)=\sI\) if \(i(P)\in S(P)\) and
\(\mathsf X(P)=\sO\) otherwise. If the next touch at \(e(P)<\infty\)
is incoming, respectively outgoing, then
\(\mathsf Y(P)=\sI\), respectively \(\mathsf Y(P)=\sO\). If there is
no later touch, then \(e(P)=\infty\) and
\(\mathsf Y(P)=\sE\). The family of all full patches is denoted by
\(\cP\).
\end{definition}

\begin{definition}[Finite-horizon patch families]
\label{def:finite-patch-families}
For \(T<\infty\), truncate every full patch intersecting
\([0,T]\) at time \(T\). The patches ending at or before \(T\) form
the bulk family \(\cB_T\), while those truncated by \(T\) form the end
family \(\cE_T\). Set
\[
  \cP_T=\cB_T\cup\cE_T.
\]
Every end patch is based at a distinct site of the interaction cone
\(\Cone_T\).
\end{definition}

\subsection{Reference patch laws and consistency}

\begin{definition}[Reference and consistent patch laws]
\label{def:consistent-patch-law}
Fix a labeled patch \(P\), and abbreviate
\[
  i=i(P),\qquad s=s(P),\qquad e=e(P),\qquad S=S(P).
\]
Under the reference law \(\pP_P\), the interactions outgoing from \(i\)
in \((s,e)\) are the restrictions of the independent Poisson clocks
in \cref{con:dual-graphical}. If \(\mathsf X(P)=\sO\), the hidden kind
of the interaction at \(s\) satisfies
\begin{equation}
  \pP_P(\alpha=\delta)
  =
  \frac{\delta_i(S)}{\delta_i(S)+\beta_i(S)},
  \qquad
  \pP_P(\alpha=\beta)
  =
  \frac{\beta_i(S)}{\delta_i(S)+\beta_i(S)}.
  \label{eq:boundary-kind-law}
\end{equation}

These data generate a one-site active indicator \(X^P\). An incoming
patch starts active. An outgoing patch starts inactive after a split
and active after a birth. Subsequently, an empty-target death or a
split makes the site inactive, while a birth leaves it active,
provided its source is active.

The consistency event \(\operatorname{Con}(P)\) requires:
\begin{enumerate}[(i)]
  \item every nonempty-target interaction strictly inside \(P\) has
  inactive source;
  \item if \(\mathsf Y(P)=\sO\), the source is active immediately
  before \(e(P)\).
\end{enumerate}
For every realized finite-horizon or full patch,
\(\pP_P(\operatorname{Con}(P))>0\) almost surely; an infinite labeled
patch with zero normalizer has zero probability of appearing in the
successful skeleton. Define
\[
  \pP_P^{\mathrm{con}}(\,\cdot\,)
  =
  \pP_P(\,\cdot\mid\operatorname{Con}(P)).
\]
\end{definition}

\begin{theorem}[Finite-horizon patch factorization]
\label{thm:patch-factorization}
Fix \(T<\infty\). Conditional on \(\cG_T\), the patch interaction data
are independent, and the conditional law in
\(P\in\cP_T\) is \(\pP_P^{\mathrm{con}}\). Thus, for bounded
measurable \(f_P\),
\begin{equation}
  \E\left[
    \prod_{P\in\cP_T}f_P
    \,\middle|\,\cG_T
  \right]
  =
  \prod_{P\in\cP_T}
  \E_P^{\mathrm{con}}[f_P].
  \label{eq:patch-factorization}
\end{equation}
\end{theorem}

\begin{proof}
Fix a realization \(g\) of the finite skeleton. It determines a
finite labeled family \(\cP_T(g)\) and finitely many boundary points.
Disintegrating the ordered Poisson points at these locations, the
independent increments on the disjoint source-time intervals give
\[
  \bigotimes_{P\in\cP_T(g)}\pP_P,
\]
and the hidden mark at every outgoing boundary has law
\eqref{eq:boundary-kind-law}.

Relative to the fixed boundary points, the event that \(g\) is the
complete successful skeleton is
\[
  \bigcap_{P\in\cP_T(g)}\operatorname{Con}(P).
  \tag{*}
\]
One implication follows from the definition. For the converse, read
the boundary interactions chronologically. The local indicators agree
with the global active state between boundary times; interior
consistency rules out a missing successful interaction; terminal
consistency activates every recorded outgoing boundary; and that
boundary starts precisely the prescribed source and target patches.
Induction reconstructs the global active process and proves \((*)\).

Conditioning the product reference law on \((*)\) gives
\[
  \frac{
    \prod_P
    \ind(\operatorname{Con}(P))\,\pP_P(\dd\Sigma_P)
  }{
    \prod_P\pP_P(\operatorname{Con}(P))
  }
  =
  \bigotimes_P\pP_P^{\mathrm{con}}(\dd\Sigma_P),
\]
which is \eqref{eq:patch-factorization}. Only finitely many
source-time strips are involved.
\end{proof}

\begin{remark}[Finite horizon]
\label{rem:finite-horizon-factorization}
\Cref{thm:patch-factorization} does not condition on an all-time
skeleton. Infinite patch factors used in
\cref{thm:common-limit} are defined by endpoint limits of
finite-horizon factors.
\end{remark}
